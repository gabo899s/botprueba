const axios = require('axios');
const { CONFIG } = require('../index.js');

class PRCClient {
    constructor() {
        if (!CONFIG.PRC_API_URL || !CONFIG.PRC_SERVER_KEY) {
            console.warn('⚠️ PRC API no configurada. Comandos ER:LC no funcionarán.');
        }
        
        // URL BASE CORREGIDA: Incluye /v1 según la documentación oficial
        const baseURL = (CONFIG.PRC_API_URL || 'https://api.policeroleplay.community') + '/v1';
        
        console.log(`🔧 Configurando API PRC en: ${baseURL}`);
        
        this.client = axios.create({
            baseURL: baseURL,
            headers: {
                'Authorization': CONFIG.PRC_AUTH_KEY ? `Bearer ${CONFIG.PRC_AUTH_KEY}` : '',
                // HEADER CORREGIDO: Usar minúsculas como en la documentación oficial
                'server-key': CONFIG.PRC_SERVER_KEY || '',
                'Content-Type': 'application/json',
                'User-Agent': 'ERLC-Discord-Bot/1.0'
            },
            timeout: 15000
        });
        
        this.lastRequest = 0;
        this.rateLimitDelay = CONFIG.RATE_LIMIT_DELAY || 1200; // 1.2 segundos por seguridad
        this.requestQueue = [];
        this.isProcessingQueue = false;
        
        console.log('✅ Cliente PRC API configurado correctamente');
    }
    
    // ========== SISTEMA DE COLA PARA RATE LIMITING ==========
    
    async throttledRequest(endpoint, method = 'GET', data = null) {
        return new Promise((resolve, reject) => {
            this.requestQueue.push({ endpoint, method, data, resolve, reject });
            this.processQueue();
        });
    }
    
    async processQueue() {
        if (this.isProcessingQueue || this.requestQueue.length === 0) return;
        
        this.isProcessingQueue = true;
        
        while (this.requestQueue.length > 0) {
            const now = Date.now();
            const timeSinceLastRequest = now - this.lastRequest;
            
            if (timeSinceLastRequest < this.rateLimitDelay) {
                await new Promise(resolve => 
                    setTimeout(resolve, this.rateLimitDelay - timeSinceLastRequest)
                );
            }
            
            const request = this.requestQueue.shift();
            
            try {
                const response = await this.makeRequest(
                    request.endpoint, 
                    request.method, 
                    request.data
                );
                request.resolve(response);
            } catch (error) {
                request.reject(error);
            }
            
            this.lastRequest = Date.now();
        }
        
        this.isProcessingQueue = false;
    }
    
    // ========== MÉTODO PRINCIPAL PARA HACER PETICIONES ==========
    
    async makeRequest(endpoint, method, data) {
        try {
            const response = await this.client({
                method,
                url: endpoint,
                data
            });
            
            console.log(`✅ API PRC [${method} ${endpoint}]: ${response.status}`);
            return response.data;
            
        } catch (error) {
            const errorDetails = {
                status: error.response?.status,
                statusText: error.response?.statusText,
                data: error.response?.data,
                endpoint,
                method,
                headers: error.config?.headers ? {
                    'server-key': '***REDACTED***', // Ocultamos la clave por seguridad
                    'content-type': error.config.headers['Content-Type']
                } : null
            };
            
            console.error(`❌ API PRC Error [${method} ${endpoint}]:`, {
                status: errorDetails.status,
                statusText: errorDetails.statusText,
                message: errorDetails.data?.message
            });
            
            // Manejo específico de errores comunes
            if (error.response?.status === 401) {
                throw new Error('API Key inválida o expirada');
            } else if (error.response?.status === 403) {
                throw new Error('Sin permisos para ejecutar este comando');
            } else if (error.response?.status === 404) {
                throw new Error(`Endpoint no encontrado: ${endpoint}. Verifica la URL base: ${this.client.defaults.baseURL}`);
            } else if (error.response?.status === 429) {
                throw new Error('Rate limit excedido. Espera antes de enviar más comandos');
            } else if (error.code === 'ECONNREFUSED') {
                throw new Error('API no disponible. Servidor ER:LC podría estar offline');
            } else if (error.code === 'ETIMEDOUT') {
                throw new Error('Timeout. La API no respondió a tiempo');
            }
            
            throw new Error(error.response?.data?.message || error.message || 'Error desconocido');
        }
    }
    
    // ========== MÉTODO PARA ENVIAR COMANDOS ER:LC ==========
    
    async enviarComando(comandoTexto) {
        // Endpoint correcto según documentación oficial
        const endpoint = '/server/command';
        
        console.log(`📤 Enviando comando ER:LC: "${comandoTexto}"`);
        console.log(`   URL completa: ${this.client.defaults.baseURL}${endpoint}`);
        
        const payload = {
            command: comandoTexto,
            timestamp: new Date().toISOString(),
            source: 'discord_bot'
        };
        
        return this.throttledRequest(endpoint, 'POST', payload);
    }
    
    // ========== MÉTODOS AUXILIARES ==========
    
    async verificarConexion() {
        try {
            await this.throttledRequest('/server', 'GET');
            return { 
                connected: true, 
                message: '✅ Conectado a API PRC',
                url: this.client.defaults.baseURL
            };
        } catch (error) {
            return { 
                connected: false, 
                message: `❌ No conectado: ${error.message}`,
                url: this.client.defaults.baseURL
            };
        }
    }
    
    async obtenerEstadoServidor() {
        return this.throttledRequest('/server', 'GET');
    }
    
    async obtenerJugadores() {
        return this.throttledRequest('/server/players', 'GET');
    }
    
    async obtenerVehiculos() {
        return this.throttledRequest('/server/vehicles', 'GET');
    }
    
    async obtenerBans() {
        return this.throttledRequest('/server/bans', 'GET');
    }
    
    // Método para diagnóstico rápido
    async diagnostico() {
        console.log('\n🔍 DIAGNÓSTICO API PRC:');
        console.log(`   URL Base: ${this.client.defaults.baseURL}`);
        console.log(`   Server-Key: ${CONFIG.PRC_SERVER_KEY ? '✅ Configurada' : '❌ NO configurada'}`);
        console.log(`   Auth Key: ${CONFIG.PRC_AUTH_KEY ? '✅ Configurada' : '⚠️  NO configurada'}`);
        
        try {
            const estado = await this.verificarConexion();
            console.log(`   Conexión: ${estado.message}`);
            
            if (estado.connected) {
                const infoServer = await this.obtenerEstadoServidor();
                console.log(`   Servidor: ${infoServer.name || 'Desconocido'}`);
                console.log(`   Jugadores: ${infoServer.playerCount || 0}/${infoServer.maxPlayers || 0}`);
            }
        } catch (error) {
            console.log(`   ❌ Error diagnóstico: ${error.message}`);
        }
        
        console.log('');
    }
}

module.exports = new PRCClient();
