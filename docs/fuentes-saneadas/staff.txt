        function formatDate(dateString) {
            if (!dateString) return 'N/A';
            const date = new Date(dateString);
            return date.toLocaleDateString('es-ES', {
                day: '2-digit',
                month: '2-digit',
                year: 'numeric',
                hour: '2-digit',
                minute: '2-digit'
            });
        }

        function formatTime(seconds) {
            const hrs = Math.floor(seconds / 3600);
            const mins = Math.floor((seconds % 3600) / 60);
            const secs = seconds % 60;
            return `${hrs.toString().padStart(2, '0')}:${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
        }

        async function checkAuth() {
            const urlParams = new URLSearchParams(window.location.search);
            const token = "VALOR_RETIRADO";
            
            if (!token) {
                showLogin();
                return;
            }

            try {
                const response = await fetch('/api/staff/me', {
                    headers: { 'Authorization': `Bearer ${token}` }
                });

                if (response.ok) {
                    const data = await response.json();
                    if (data.success) {
                        currentUser = data.user;
                        currentToken = "VALOR_RETIRADO";
                        localStorage.setItem('staff_token', token);
                        window.history.replaceState({}, '', '/staff');
                        loadStaffData();
                        showDashboard();
                    } else {
                        showLogin();
                    }
                } else {
                    localStorage.removeItem('staff_token');
                    showLogin();
                }
            } catch (error) {
                console.error('Error verificando autenticación:', error);
                showLogin();
            }
        }

                                       shiftData.estado === 'descansando' ? 'Descansando' : 'Inactivo';

        function startShiftTimer() {
            if (shiftInterval) clearInterval(shiftInterval);
            
            shiftInterval = setInterval(() => {
                if (shiftStartTime) {
                    const now = new Date();
                    const diff = Math.floor((now - shiftStartTime) / 1000);
                    document.getElementById('shift-timer').textContent = formatTime(diff);
                }
            }, 1000);
        }

                    return;
                }
                
                await searchHistoryUsers(query);
            }, 300));
        }

        function debounce(func, wait) {
            let timeout;
            return function executedFunction(...args) {
                const later = () => {
                    clearTimeout(timeout);
                    func(...args);
                };
                clearTimeout(timeout);
                timeout = setTimeout(later, wait);
            };
        }

        function displayUserLogs(data) {
            const container = document.getElementById('logs-list');
            const logs = [...(data.sanciones || []), ...(data.baneos || [])];
            currentLogs = logs;
            
            if (logs.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-8 text-slate-500">
                        <i data-lucide="file-text" class="w-8 h-8 mx-auto mb-2 opacity-30"></i>
                        <p class="text-sm">No hay logs para este usuario</p>
                    </div>
                `;
                document.getElementById('logs-count').textContent = '0 logs';
                return;
            }
            
            // Ordenar por fecha (más reciente primero)
            logs.sort((a, b) => new Date(b.fecha || b.start) - new Date(a.fecha || a.start));
            
            container.innerHTML = '';
            logs.forEach((log, index) => {
                const logElement = createLogElement(log, index);
                container.appendChild(logElement);
            });
            
            document.getElementById('logs-count').textContent = `${logs.length} logs`;
        }

        function createLogElement(log, index) {
            const element = document.createElement('div');
            element.className = `log-item p-4 ${log.estado === 'apelada' ? 'log-appealed' : log.estado ? 'log-sancion' : 'log-ban'}`;
            
            const isSanction = log.estado !== undefined;
            const logType = isSanction ? 'Sanción' : 'Baneo';
            const logDate = formatDate(log.fecha || log.start);
            const logStatus = isSanction ? 
                (log.estado === 'activa' ? 'Activa' : 'Apelada') : 
                (log.active ? 'Activo' : 'Finalizado');
            
            let durationText = '';
            if (!isSanction && log.days) {
                durationText = log.days === 9999 ? 'Permanente' : `${log.days} día(s)`;
            }
            
            element.innerHTML = `
                <div class="flex justify-between items-start mb-2">
                    <div>
                        <div class="font-bold text-white">${logType} #${index + 1}</div>
                        <div class="text-xs text-slate-400">${logDate}</div>
                    </div>
                    <div class="flex items-center gap-2">
                        <span class="text-xs px-2 py-1 rounded-full ${logStatus === 'Activa' || logStatus === 'Activo' ? 'badge-active' : logStatus === 'Apelada' ? 'badge-loan' : 'badge-inactive'}">
                            ${logStatus}
                        </span>
                        <div class="relative">
                            <button onclick="toggleLogMenu('log-menu-${index}')" class="p-1 hover:bg-white/10 rounded">
                                <i data-lucide="more-vertical" class="w-4 h-4"></i>
                            </button>
                            <div id="log-menu-${index}" class="three-dots-menu">
                                ${isSanction ? `
                                    <div class="menu-item" onclick="appealLog('${log.id}')">
                                        <i data-lucide="scale" class="w-4 h-4"></i>
                                        Apelar
                                    </div>
                                ` : `
                                    <div class="menu-item" onclick="unbanLog('${log.id}')">
                                        <i data-lucide="unlock" class="w-4 h-4"></i>
                                        Desbanear
                                    </div>
                                `}
                                <div class="menu-item" onclick="editLog('${log.id}', ${isSanction})">
                                    <i data-lucide="edit" class="w-4 h-4"></i>
                                    Editar
                                </div>
                                <div class="menu-item danger" onclick="confirmDeleteLog('${log.id}', ${isSanction})">
                                    <i data-lucide="trash-2" class="w-4 h-4"></i>
                                    Eliminar
                                </div>
                            </div>
                        </div>
                    </div>
                </div>
                <div class="text-sm text-slate-300 mb-2">${log.motivo || 'Sin motivo especificado'}</div>
                ${durationText ? `<div class="text-xs text-slate-500">Duración: ${durationText}</div>` : ''}
            `;
            
            return element;
        }

        async function createLog() {
            if (!selectedUser) {
                showToast('Primero selecciona un usuario', 'error');
                return;
            }
            
            const logType = document.getElementById('log-type').value;
            const reason = document.getElementById('log-reason').value;
            
            if (!logType || !reason) {
                showToast('Por favor completa todos los campos', 'error');
                return;
            }
            
            try {
                const endpoint = logType.startsWith('ban_') ? '/api/staff/banear' : '/api/staff/sancionar';
                const dias = logType.includes('ban_') ? 
                    (logType === 'ban_1' ? 1 : logType === 'ban_3' ? 3 : logType === 'ban_7' ? 7 : 9999) : null;
                
                const requestData = logType.startsWith('ban_') ? {
                    discordId: selectedUser.discordId,
                    dias: dias,
                    motivo: reason,
                    staffId: currentUser.id
                } : {
                    discordId: selectedUser.discordId,
                    tipo: logType,
                    motivo: reason,
                    staffId: currentUser.id
                };
                
                const response = await fetch(endpoint, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'Authorization': `Bearer ${currentToken}`
                    },
                    body: JSON.stringify(requestData)
                });

                const data = await response.json();
                
                if (data.success) {
                    showToast('Log creado exitosamente', 'success');
                    document.getElementById('log-form').reset();
                    await loadUserLogs(selectedUser.discordId);
                    
                    // Actualizar estadísticas
                    await loadActivityStats();
                } else {
                    showToast(data.message || 'Error al crear log', 'error');
                }
            } catch (error) {
                console.error('Error creando log:', error);
                showToast('Error de conexión', 'error');
            }
        }

            const totalHours = shiftData.totalHoras || 0;
            document.getElementById('total-hours').textContent = `${totalHours.toFixed(1)}h`;
            
            const avgShift = shiftData.historial?.length > 0 ? 
                (totalHours / shiftData.historial.length).toFixed(1) : 0;
            document.getElementById('avg-shift').textContent = `${avgShift}h`;
            
            const historyContainer = document.getElementById('shifts-history');
            historyContainer.innerHTML = '';
            
            if (!shiftData.historial || shiftData.historial.length === 0) {
                historyContainer.innerHTML = `
                    <div class="text-center py-8 text-slate-500">
                        <i data-lucide="clock" class="w-8 h-8 mx-auto mb-2 opacity-30"></i>
                        <p class="text-sm">No hay shifts registrados</p>
                    </div>
                `;
                return;
            }
            
            shiftData.historial.slice(0, 10).forEach((shift, index) => {
                const shiftElement = document.createElement('div');
                shiftElement.className = 'p-3 bg-white/5 rounded-lg';
                
                const inicio = formatDate(shift.inicio);
                const fin = shift.fin ? formatDate(shift.fin) : 'En curso';
                const duracion = shift.duracionHoras ? `${shift.duracionHoras}h` : 'N/A';
                
                shiftElement.innerHTML = `
                    <div class="flex justify-between items-center mb-1">
                        <div class="font-medium text-white">Shift #${shiftData.historial.length - index}</div>
                        <div class="text-sm text-slate-400">${duracion}</div>
                    </div>
                    <div class="text-xs text-slate-500">
                        <div>Inicio: ${inicio}</div>
                        <div>Fin: ${fin}</div>
                    </div>
                `;
                
                historyContainer.appendChild(shiftElement);
            });
        }
