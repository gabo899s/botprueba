        // Función para mostrar notificaciones
        function showToast(msg, type = "info") {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            toast.className = `toast ${type}`;
            toast.innerHTML = `
                <div class="flex items-center gap-3">
                    <i data-lucide="${type === 'error' ? 'alert-circle' : type === 'warning' ? 'alert-triangle' : 'check-circle'}" class="w-5 h-5"></i>
                    <span class="text-sm font-bold tracking-tight">${msg}</span>
                </div>
                <button onclick="this.parentElement.remove()" class="ml-4 p-1 hover:bg-white/10 rounded-full transition-colors">
                    <i data-lucide="x" class="w-4 h-4 text-white/50"></i>
                </button>
            `;
            container.appendChild(toast);
            lucide.createIcons();
            
            setTimeout(() => { if(toast.parentElement) toast.remove(); }, 6000);
        }

        // Variables globales
        let currentUser = null;
        let selectedStaff = null;
        let searchTimeout = null;

        // Verificar autenticación al cargar
        async function checkAuth() {
            const urlParams = new URLSearchParams(window.location.search);
            const token = "VALOR_RETIRADO";
            const error = urlParams.get('error');
            
            // Manejar errores de autenticación
            if (error === 'no_permission') {
                showToast('No tienes los roles necesarios para acceder al panel de administración', 'error');
                // Limpiar la URL
                window.history.replaceState({}, '', '/admin');
            } else if (error === 'auth_failed') {
                showToast('Error en la autenticación con Discord', 'error');
                window.history.replaceState({}, '', '/admin');
            }
            
            if (token) {
                localStorage.setItem('admin_token', token);
                // Limpiar la URL
                window.history.replaceState({}, '', '/admin');
            }
            
            const storedToken = "VALOR_RETIRADO";
            
            if (!storedToken) {
                // Mostrar vista de login
                document.getElementById('view-login').classList.remove('hidden');
                document.getElementById('view-dashboard').classList.add('hidden');
                return;
            }
            
            // Verificar token
            try {
                // Decodificar el token para obtener info básica
                const payload = JSON.parse(atob(storedToken.split('.')[1]));
                currentUser = payload;
                
                document.getElementById('view-login').classList.add('hidden');
                document.getElementById('view-dashboard').classList.remove('hidden');
                updateUserInfo();
                
            } catch (err) {
                console.error('Error verificando token:"VALOR_RETIRADO";
                localStorage.removeItem('admin_token');
                checkAuth();
            }
        }

        function updateUserInfo() {
            if (!currentUser) return;
            
            document.getElementById('user-name').textContent = 
                currentUser.discordUsername || 'Administrador';
            
            document.getElementById('user-role').textContent = 
                currentUser.roles ? `(${currentUser.roles.length} roles)` : 'Administrador';
            
            const avatarUrl = currentUser.discordAvatar || 
                            'https://cdn.discordapp.com/embed/avatars/0.png';
            document.getElementById('user-avatar').src = avatarUrl;
        }

        async function doLogout() {
            localStorage.removeItem('admin_token');
            currentUser = null;
            checkAuth();
            showToast("Sesión cerrada correctamente", "info");
        }

        // Búsqueda
        document.getElementById('search-input')?.addEventListener('input', function(e) {
            const query = e.target.value.trim();
            
            // Limpiar timeout anterior
            if (searchTimeout) {
                clearTimeout(searchTimeout);
            }
            
            if (!query || query.length < 1) {
                return;
            }
            
            // Esperar 300ms después de que el usuario deje de escribir
            searchTimeout = setTimeout(async () => {
                await handleSearch(query);
            }, 300);
        });

        async function handleSearch(query) {
            const token = "VALOR_RETIRADO";
            if (!token) {
                showToast('Sesión expirada. Por favor, ingrese nuevamente.', 'error');
                checkAuth();
                return;
            }

            try {
                const res = await fetch(`/api/admin/search?q=${encodeURIComponent(query)}`, {
                    headers: {
                        'Authorization': `Bearer ${token}`
                    }
                });
                
                const data = await res.json();
                
                if (data.success) {
                    selectedStaff = data.data;
                    renderStaff(data.data);
                    showToast("Staff encontrado correctamente", "success");
                } else {
                    showToast(data.message, "error");
                }
            } catch (err) {
                showToast("Error de conexión con el servidor", "error");
            }
        }

        function renderStaff(staff) {
            document.getElementById('dashboard-empty').classList.add('hidden');
            document.getElementById('staff-content').classList.remove('hidden');

            // Avatar
            const avatarUrl = staff.discordAvatar || 
                            `https://cdn.discordapp.com/avatars/${staff.discordId}/${staff.avatar}.png?size=256` ||
                            'https://cdn.discordapp.com/embed/avatars/0.png';
            document.getElementById('staff-avatar').src = avatarUrl;
            
            // Nombre y placa
            document.getElementById('staff-name').textContent = staff.nombre || staff.discordUsername || 'Sin nombre';

            const details = {

                telefono: staff.telefono,
                correo: staff.correo
            };
            
            for (const [key, value] of Object.entries(details)) {
                if (value) {
                    const label = {
                        discordId: 'ID Discord',
                        discordUsername: 'Usuario Discord',

                    detailsDiv.innerHTML += `
                        <div class="flex justify-between border-b border-white/10 pb-2 items-end">
                            <span class="text-white/50 font-bold text-xs uppercase tracking-wider">${label}</span>
                            <span class="text-white text-sm font-medium">${value}</span>
                        </div>`;
                }
            }

            // Estadísticas
            document.getElementById('stats-sanciones').textContent = staff.sancionesHechas?.length || 0;

                            <div class="flex justify-between text-sm">

                            <div class="flex justify-between text-sm">

            } else {

            }

            // Sanciones Recientes
            const sancionesDiv = document.getElementById('staff-sanciones');
            if (staff.sancionesHechas && staff.sancionesHechas.length > 0) {
                sancionesDiv.innerHTML = '';
                
                // Mostrar solo las 5 más recientes
                const sancionesRecientes = staff.sancionesHechas.slice(0, 5);
                
                sancionesRecientes.forEach(s => {
                    const nivelColors = {
                        1: 'bg-yellow-500/20 text-yellow-400',
                        2: 'bg-orange-500/20 text-orange-400',
                        3: 'bg-red-500/20 text-red-400',
                        4: 'bg-purple-500/20 text-purple-400',
                        5: 'bg-pink-500/20 text-pink-400'
                    };
                    
                    sancionesDiv.innerHTML += `
                        <div class="sancion-card p-5 rounded-2xl">
                            <div class="flex justify-between items-start mb-3">
                                <div>
                                    <div class="text-xs font-black text-red-500 uppercase mb-1 tracking-widest">SANCIÓN NIVEL ${s.nivel}</div>
                                    <div class="text-sm text-white font-medium">${s.motivo}</div>
                                </div>
                                <div class="text-xs px-2 py-1 rounded ${nivelColors[s.nivel] || 'bg-gray-500/20 text-gray-400'}">Nivel ${s.nivel}</div>
                            </div>
                            <div class="flex justify-between items-center text-xs">
                                <div class="text-white/60">
                                    <i data-lucide="calendar" class="w-3 h-3 inline mr-1"></i>
                                    ${new Date(s.fecha).toLocaleDateString()}
                                </div>
                                <div class="text-white/60">
                                    <i data-lucide="user" class="w-3 h-3 inline mr-1"></i>
                                    Afectado: ${s.userId || 'N/A'}
                                </div>
                                <div class="text-white/60">
                                    Estado: <span class="${s.estado === 'activa' ? 'text-red-400' : 'text-green-400'}">${s.estado}</span>
                                </div>
                            </div>
                        </div>
                    `;
                });
                
                // Mostrar contador total
                if (staff.sancionesHechas.length > 5) {
                    sancionesDiv.innerHTML += `
                        <div class="text-center text-white/60 text-sm mt-4">
                            +${staff.sancionesHechas.length - 5} sanciones más...
                        </div>
                    `;
                }
            } else {
                sancionesDiv.innerHTML = `
                    <div class="p-8 text-center border border-dashed border-white/20 rounded-2xl">
                        <i data-lucide="gavel" class="w-8 h-8 mx-auto mb-2 text-white/30"></i>
                        <p class="text-white/50 text-sm">No ha realizado sanciones</p>
                    </div>
                `;
            }

            // Tickets Atendidos
            const ticketsDiv = document.getElementById('staff-tickets');
            if (staff.tickets) {
                ticketsDiv.innerHTML = `
                    <div class="ticket-card p-5 rounded-2xl">
                        <div class="grid grid-cols-2 gap-4 mb-4">
                            <div>
                                <div class="text-xs text-white/60 mb-1">Total atendidos</div>
                                <div class="text-3xl font-black text-cyan-400">${staff.tickets.count || 0}</div>
                            </div>
                            <div>
                                <div class="text-xs text-white/60 mb-1">Calificación promedio</div>
                                <div class="text-3xl font-black text-yellow-400">
                                    ${staff.tickets.stars && staff.tickets.stars.length > 0 ? 
                                        (staff.tickets.stars.reduce((a, b) => a + b, 0) / staff.tickets.stars.length).toFixed(1) + ' ★' : 
                                        'N/A'}
                                </div>
                            </div>
                        </div>
                        
                        ${staff.tickets.categories ? `
                            <div class="mt-4">
                                <div class="text-xs text-white/60 mb-2 uppercase tracking-wider">Distribución por categoría</div>
                                <div class="space-y-2">
                                    ${Object.entries(staff.tickets.categories).map(([cat, count]) => `
                                        <div class="flex items-center justify-between">
                                            <span class="text-sm text-white/80">${cat}</span>
                                            <div class="flex items-center gap-2">
                                                <div class="w-24 bg-white/10 rounded-full h-2">
                                                    <div class="bg-cyan-500 h-2 rounded-full" style="width: ${(count / staff.tickets.count) * 100}%"></div>
                                                </div>
                                                <span class="text-sm font-bold text-white">${count}</span>
                                            </div>
                                        </div>
                                    `).join('')}
                                </div>
                            </div>
                        ` : ''}
                        
                        ${staff.tickets.stars && staff.tickets.stars.length > 0 ? `
                            <div class="mt-6 pt-4 border-t border-white/10">
                                <div class="text-xs text-white/60 mb-2 uppercase tracking-wider">Historial de calificaciones</div>
                                <div class="flex flex-wrap gap-1">
                                    ${staff.tickets.stars.map(star => `
                                        <div class="text-yellow-400">
                                            ${'★'.repeat(star)}${'☆'.repeat(5-star)}
                                        </div>
                                    `).join('')}
                                </div>
                            </div>
                        ` : ''}
                    </div>
                `;
            } else {
                ticketsDiv.innerHTML = `
                    <div class="p-8 text-center border border-dashed border-white/20 rounded-2xl">
                        <i data-lucide="ticket" class="w-8 h-8 mx-auto mb-2 text-white/30"></i>
                        <p class="text-white/50 text-sm">No ha atendido tickets</p>
                    </div>
                `;
            }

            // Calificaciones
            const calificacionesDiv = document.getElementById('staff-calificaciones');
            if (staff.calificaciones) {
                // Ajusta según la estructura real de calificaciones_staff.json
                if (typeof staff.calificaciones === 'object' && Object.keys(staff.calificaciones).length > 0) {
                    calificacionesDiv.innerHTML = '';
                    
                    for (const [key, calificacion] of Object.entries(staff.calificaciones)) {
                        if (calificacion && typeof calificacion === 'object') {
                            calificacionesDiv.innerHTML += `
                                <div class="calificacion-card p-5 rounded-2xl">
                                    <div class="flex justify-between items-start mb-3">
                                        <div>
                                            <div class="text-sm font-bold text-white">${calificacion.calificador || 'Anónimo'}</div>
                                            <div class="text-xs text-white/60 mt-1">${new Date(calificacion.fecha || key).toLocaleDateString()}</div>
                                        </div>
                                        <div class="text-2xl font-black text-yellow-400">
                                            ${'★'.repeat(calificacion.estrellas || 0)}${'☆'.repeat(5 - (calificacion.estrellas || 0))}
                                        </div>
                                    </div>
                                    ${calificacion.comentario ? `
                                        <div class="mt-3 p-3 bg-white/5 rounded-lg">
                                            <p class="text-sm text-white/80">"${calificacion.comentario}"</p>
                                        </div>
                                    ` : ''}
                                </div>
                            `;
                        }
                    }
                } else {
                    calificacionesDiv.innerHTML = `
                        <div class="p-8 text-center border border-dashed border-white/20 rounded-2xl">
                            <i data-lucide="star" class="w-8 h-8 mx-auto mb-2 text-white/30"></i>
                            <p class="text-white/50 text-sm">No hay calificaciones registradas</p>
                        </div>
                    `;
                }
            } else {
                calificacionesDiv.innerHTML = `
                    <div class="p-8 text-center border border-dashed border-white/20 rounded-2xl">
                        <i data-lucide="star" class="w-8 h-8 mx-auto mb-2 text-white/30"></i>
                        <p class="text-white/50 text-sm">No hay calificaciones registradas</p>
                    </div>
                `;
            }
            
            lucide.createIcons();
        }

        // Inicializar
