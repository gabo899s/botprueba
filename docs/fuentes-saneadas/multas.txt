async function cargarCedulas() {
    try {
        const data = await fs.readFile(CEDULAS_PATH, 'utf8');
        return JSON.parse(data);
    } catch (error) {
        console.error('Error cargando cédulas:', error);
        return {};
    }
}

async function cargarMultas() {
    try {
        // Verificar si el archivo existe
        try {
            await fs.access(MULTAS_PATH);
        } catch {
            // Si no existe, crearlo con un objeto vacío
            console.log(`[DEPURACIÓN] Archivo multas.json no existe, creando...`);
            await fs.writeFile(MULTAS_PATH, JSON.stringify({}, null, 2));
            return {};
        }
        
        const data = await fs.readFile(MULTAS_PATH, 'utf8');
        console.log(`[DEPURACIÓN] Archivo multas.json leído, tamaño: ${data.length} bytes`);
        
        // Verificar si el archivo está vacío o tiene contenido inválido
        if (!data.trim()) {
            console.log(`[DEPURACIÓN] Archivo multas.json está vacío`);
            return {};
        }
        
        const parsed = JSON.parse(data);
        console.log(`[DEPURACIÓN] Multas cargadas: ${Object.keys(parsed).length}`);
        return parsed;
    } catch (error) {
        console.error('Error cargando multas:', error);
        console.error('Detalles del error:', error.stack);
        return {};
    }
}

    let diasHabiles = 0;
    let fecha = new Date(horaCR);
    
    while (diasHabiles < 3) {
        fecha.setDate(fecha.getDate() + 1);
        const dia = fecha.getDay();
        if (dia !== 0 && dia !== 6) { // No es domingo (0) ni sábado (6)

// Función para calcular el monto automático basado en los artículos seleccionados
function calcularMontoAutomatico(articulosTexto) {
    let montoTotal = 0;
    
    // Buscar todos los números de artículos en el texto
    // Formato esperado: "Art.1 (...)" o solo el número del artículo
    const matches = articulosTexto.match(/Art\.(\d+)/gi);
    
    if (matches) {
        matches.forEach(match => {
            const numeroArticulo = parseInt(match.replace(/Art\./i, ''));
            const articulo = ARTICULOS.find(art => art.numero === numeroArticulo);
            if (articulo) {
                montoTotal += articulo.monto;
            }
        });
    }
    
    return montoTotal;
}

            const balance = await obtenerBalanceUnbelievabot(usuarioId);
            console.log(`[DEPURACIÓN] Balance del usuario: $${balance}`);
            
            if (balance < total) {
                await interaction.reply({
                    components: [formatoError(`No tienes suficiente dinero. Necesitas $${total}, tienes $${balance}.`)],
                    flags: MessageFlags.IsComponentsV2,
                    ephemeral: true
                });
                return;
            }
            
            console.log(`[DEPURACIÓN] Intentando cobrar $${total} al usuario ${usuarioId}...`);
            const exito = await modificarBalanceUnbelievabot(usuarioId, -total);
            
            if (!exito) {
                await interaction.reply({
                    components: [formatoError('Error al procesar el pago. Intenta más tarde.')],
                    flags: MessageFlags.IsComponentsV2,
                    ephemeral: true
                });
                return;
            }
            
            console.log(`[DEPURACIÓN] Pago exitoso. Actualizando estado de multas...`);
            
            const fechaPago = new Date().toISOString();
            for (const multa of multasAPagar) {
                multa.estado = 'pagada';
                multa.pagadoEn = fechaPago;
                multa.metodoPago = 'manual';
                multas[multa.id] = multa;
                console.log(`[DEPURACIÓN] Multa ${multa.id} marcada como pagada.`);
            }
            
            await guardarMultas(multas);
            
            seleccionesTemporales.delete(usuarioId);
            
            // Enviar al canal de multas cuando se PAGA
            const canalMultas = interaction.client.channels.cache.get(CANAL_MULTAS_ID);
            if (canalMultas) {
                const cedulas = await cargarCedulas();
                const cedulaUsuario = cedulas[usuarioId] || {};
                
                const containerCanal = new ContainerBuilder()
                    .addTextDisplayComponents(
                        new TextDisplayBuilder()
                            .setContent('✅ **Multa(s) Pagada(s)**')
                    )
                    .addSeparatorComponents(
                        new SeparatorBuilder()
                            .setDivider(true)
                            .setSpacing(SeparatorSpacingSize.Small)
                    )
                    .addTextDisplayComponents(
                        new TextDisplayBuilder()
                            .setContent(`${cedulaUsuario.nombre || 'Usuario'} ${cedulaUsuario.apellido1 || ''} (<@${usuarioId}>) ha pagado **${multasAPagar.length}** multa(s) por un total de **$${total}**.`)
                    )
                    .addSeparatorComponents(
                        new SeparatorBuilder()
                            .setDivider(true)
                            .setSpacing(SeparatorSpacingSize.Small)
                    )
                    .addTextDisplayComponents(
                        new TextDisplayBuilder()
                            .setContent('-# © Todos los derechos reservados, Chicago Real Life Roleplay | ER:LC | Spanish')
                    );
                
                await canalMultas.send({
                    components: [containerCanal],
                    flags: MessageFlags.IsComponentsV2
                });
            }
            
            const confirmacionContainer = new ContainerBuilder()
                .addTextDisplayComponents(
                    new TextDisplayBuilder()
                        .setContent('✅ **Pago Exitoso**')
                )
                .addSeparatorComponents(
                    new SeparatorBuilder()
                        .setDivider(true)
                        .setSpacing(SeparatorSpacingSize.Small)
                )
                .addTextDisplayComponents(
                    new TextDisplayBuilder()
                        .setContent(`✅ Has pagado **${multasAPagar.length}** multa(s) por un total de **$${total}**.`)
                )
                .addSeparatorComponents(
                    new SeparatorBuilder()
                        .setDivider(true)
                        .setSpacing(SeparatorSpacingSize.Small)
                )
                .addTextDisplayComponents(
                    new TextDisplayBuilder()
                        .setContent('-# © Todos los derechos reservados, Chicago Real Life Roleplay | ER:LC | Spanish')
                );
            
            await interaction.reply({
                components: [confirmacionContainer],
                flags: MessageFlags.IsComponentsV2,
                ephemeral: true
            });
        }
    }
};

const comandoVerMultas = {
    data: new SlashCommandBuilder()
        .setName('ver-multas')
        .setDescription('Muestra todas las multas registradas')
        .addUserOption(option =>
            option.setName('usuario')
                .setDescription('Filtrar por usuario (opcional)')
                .setRequired(false))
        .addStringOption(option =>
            option.setName('id_multa')
                .setDescription('Buscar por ID de multa (opcional)')
                .setRequired(false)
                .setAutocomplete(true)),
    
    async autocomplete(interaction) {
        const focusedValue = interaction.options.getFocused().toLowerCase();
        const multas = await cargarMultas();
        const cedulas = await cargarCedulas();
        
        const choices = Object.values(multas).map(multa => {
            const cedulaUsuario = cedulas[multa.usuarioId] || {};
            const cedulaOficial = cedulas[multa.oficialId] || {};
            const nombreUsuario = cedulaUsuario.nombre || 'Desconocido';
            const nombreOficial = cedulaOficial.nombre || 'Desconocido';
            
            return {
                name: `Multa - ${nombreOficial} ${nombreUsuario} $${multa.monto}`,
                value: multa.id
            };
        });
        
        const filtered = choices.filter(choice =>
            choice.name.toLowerCase().includes(focusedValue) ||
            choice.value.toLowerCase().includes(focusedValue)
        );
        
        await interaction.respond(filtered.slice(0, 25));
    },
    
    async execute(interaction) {
        console.log('=== DEPURACIÓN: VERIFICANDO INTEGRIDAD DE MULTAS ===');
        await verificarIntegridadMultas();
        
        const multas = await cargarMultas();
        const cedulas = await cargarCedulas();
        const usuarioFiltro = interaction.options.getUser('usuario');
        const idMultaFiltro = interaction.options.getString('id_multa');
        
        console.log(`[DEPURACIÓN] Total multas en sistema para /ver-multas: ${Object.keys(multas).length}`);
        
        let multasFiltradas = Object.values(multas);
        
        // Filtrar por usuario si se especificó
        if (usuarioFiltro) {
            multasFiltradas = multasFiltradas.filter(m => m.usuarioId === usuarioFiltro.id);
        }
        
        // Filtrar por ID de multa si se especificó
        if (idMultaFiltro) {
            multasFiltradas = multasFiltradas.filter(m => m.id === idMultaFiltro);
        }
        
        if (multasFiltradas.length === 0) {
            await interaction.reply({
                components: [formatoError('No se encontraron multas con los filtros especificados.')],
                flags: MessageFlags.IsComponentsV2,
                ephemeral: true
            });
            return;
        }
        
        // Formatear lista de multas
        let textoMultas = '';
        multasFiltradas.forEach((multa, index) => {
            const cedulaUsuario = cedulas[multa.usuarioId] || {};
            const cedulaOficial = cedulas[multa.oficialId] || {};
            
            textoMultas += `**Multa #${index + 1}**\n`;
            textoMultas += `ID: \`${multa.id}\`\n`;
            textoMultas += `Usuario: ${cedulaUsuario.nombre || 'Desconocido'} ${cedulaUsuario.apellido1 || ''}\n`;
            textoMultas += `Oficial: ${cedulaOficial.nombre || 'Desconocido'} ${cedulaOficial.apellido1 || ''}\n`;
            textoMultas += `Monto: $${multa.monto}\n`;
            textoMultas += `Estado: ${multa.estado ? multa.estado.toUpperCase() : 'SIN ESTADO'}\n`;
            textoMultas += `Fecha: <t:${obtenerTimestampDiscord(multa.fechaCreacion)}:R>\n`;
            textoMultas += `---\n\n`;
        });
        
        const container = new ContainerBuilder()
            .addTextDisplayComponents(
                new TextDisplayBuilder()
                    .setContent('✅ **Todas las Multas**')
            )
            .addSeparatorComponents(
                new SeparatorBuilder()
                    .setDivider(true)
                    .setSpacing(SeparatorSpacingSize.Small)
            )
            .addTextDisplayComponents(
                new TextDisplayBuilder()
                    .setContent(textoMultas)
            )
            .addSeparatorComponents(
                new SeparatorBuilder()
                    .setDivider(true)
                    .setSpacing(SeparatorSpacingSize.Small)
            )
            .addTextDisplayComponents(
                new TextDisplayBuilder()
                    .setContent('-# © Todos los derechos reservados, Chicago Real Life Roleplay | ER:LC | Spanish')
            );
        
        await interaction.reply({
            components: [container],
            flags: MessageFlags.IsComponentsV2
        });
    }
};

// ==================== SISTEMA DE RECORDATORIOS Y COBRO AUTOMÁTICO ====================

async function revisarMultasPendientes(client) {
    console.log('🔍 Revisando multas pendientes...');
    
    const multas = await cargarMultas();
    const cedulas = await cargarCedulas();
    const ahora = new Date();
    
    console.log(`[REVISIÓN] Total multas a revisar: ${Object.keys(multas).length}`);
    
    for (const [id, multa] of Object.entries(multas)) {
        if (!multa.estado || multa.estado.toLowerCase() !== 'pendiente') {
            console.log(`[REVISIÓN] Multa ${id} no está pendiente (estado: ${multa.estado}), saltando...`);
            continue;
        }
        
        console.log(`[REVISIÓN] Procesando multa ${id} para usuario ${multa.usuarioId}`);
        
        const fechaVencimiento = new Date(multa.fechaVencimiento);
        const horasRestantes = (fechaVencimiento - ahora) / (1000 * 60 * 60);
        
        console.log(`[REVISIÓN] Horas restantes: ${horasRestantes.toFixed(2)}`);
        
        // Enviar recordatorios diarios
        if (horasRestantes > 0 && horasRestantes < 24) {
            const diasRestantes = Math.ceil(horasRestantes / 24);
            
            if (!multa.recordatoriosEnviados || !multa.recordatoriosEnviados.includes(diasRestantes)) {
                const usuario = await client.users.fetch(multa.usuarioId).catch(() => {
                    console.log(`[REVISIÓN] No se pudo obtener usuario ${multa.usuarioId}`);
                    return null;
                });
                
                if (usuario) {
                    console.log(`[REVISIÓN] Enviando recordatorio a ${usuario.tag} (${diasRestantes} día(s) restantes)`);
                    
                    const recordatorioContainer = new ContainerBuilder()
                        .addTextDisplayComponents(
                            new TextDisplayBuilder()
                                .setContent('✅ **Recordatorio de Multa**')
                        )
                        .addSeparatorComponents(
                            new SeparatorBuilder()
                                .setDivider(true)
                                .setSpacing(SeparatorSpacingSize.Small)
                        )
                        .addTextDisplayComponents(
                            new TextDisplayBuilder()
                                .setContent(`Te quedan **${diasRestantes} día(s)** para pagar la multa.`)
                        )
                        .addSeparatorComponents(
                            new SeparatorBuilder()
                                .setDivider(true)
                                .setSpacing(SeparatorSpacingSize.Small)
                        )
                        .addTextDisplayComponents(
                            new TextDisplayBuilder()
                                .setContent(`Monto: $${multa.monto}\nArtículos: ${multa.articulos}`)
                        )
                        .addSeparatorComponents(
                            new SeparatorBuilder()
                                .setDivider(true)
                                .setSpacing(SeparatorSpacingSize.Small)
                        )
                        .addTextDisplayComponents(
                            new TextDisplayBuilder()
                                .setContent('-# © Todos los derechos reservados, Chicago Real Life Roleplay | ER:LC | Spanish')
                        );
                    
                    await usuario.send({
                        components: [recordatorioContainer],
                        flags: MessageFlags.IsComponentsV2
                    }).catch(error => {
                        console.log(`[REVISIÓN] Error enviando DM a ${usuario.tag}:`, error.message);
                    });
                    
                    if (!multa.recordatoriosEnviados) {
                        multa.recordatoriosEnviados = [];
                    }
                    multa.recordatoriosEnviados.push(diasRestantes);
                    multas[id] = multa;
                }
            }
        }
        
        // Cobro automático si ya venció
        if (horasRestantes <= 0) {
            console.log(`⚡ Cobrando multa automáticamente: ${id}`);
            
            const montoRecargo = Math.floor(multa.monto * 0.1);
            const montoTotal = multa.monto + montoRecargo;
            
            const balanceUsuario = await obtenerBalanceUnbelievabot(multa.usuarioId);
            console.log(`[COBRO AUTOMÁTICO] Balance usuario ${multa.usuarioId}: ${balanceUsuario}, Total a cobrar: ${montoTotal}`);
            
            if (balanceUsuario >= montoTotal) {
                const cobroExitoso = await modificarBalanceUnbelievabot(multa.usuarioId, -montoTotal);
                
                if (cobroExitoso) {
                    console.log(`[COBRO AUTOMÁTICO] Cobro exitoso de ${montoTotal} a ${multa.usuarioId}`);
                    
                    const montoOficial = Math.floor(multa.monto * 0.25);
                    const montoDirector = Math.floor(multa.monto * 0.5);
                    
                    console.log(`[COBRO AUTOMÁTICO] Distribuyendo: ${montoOficial} al oficial, ${montoDirector} al director`);
                    
                    await modificarBalanceUnbelievabot(multa.oficialId, montoOficial);
                    await modificarBalanceUnbelievabot(DIRECTOR_TRANSITO_ID, montoDirector);
                    
                    multa.estado = 'cobrada_auto';
                    multa.pagadoEn = new Date().toISOString();
                    multa.metodoPago = 'automatico';
                    multas[id] = multa;
                    
                    // Enviar al canal cuando se cobra automáticamente
                    const canalMultas = client.channels.cache.get(CANAL_MULTAS_ID);
                    if (canalMultas) {
                        const cedulaUsuario = cedulas[multa.usuarioId] || {};
                        
                        const containerCobro = new ContainerBuilder()
                            .addTextDisplayComponents(
                                new TextDisplayBuilder()
                                    .setContent('✅ **Multa Cobrada Automáticamente**')
                            )
                            .addSeparatorComponents(
                                new SeparatorBuilder()
                                    .setDivider(true)
                                    .setSpacing(SeparatorSpacingSize.Small)
                            )
                            .addTextDisplayComponents(
                                new TextDisplayBuilder()
                                    .setContent(`Estimado/a ${cedulaUsuario.apellido1 || ''} ${cedulaUsuario.apellido2 || ''} ${cedulaUsuario.apellido3 || ''} ${cedulaUsuario.nombre || 'Usuario'} (<@${multa.usuarioId}>), de parte de la División de Administración de Citaciones (CAD) se le informa que se le ha cobrado la multa con el ID (\`${id}\`) automáticamente.`)
                            )
                            .addSeparatorComponents(
                                new SeparatorBuilder()
                                    .setDivider(true)
                                    .setSpacing(SeparatorSpacingSize.Small)
                            )
                            .addTextDisplayComponents(
                                new TextDisplayBuilder()
                                    .setContent(`Monto Original: $${multa.monto}`),
                                new TextDisplayBuilder()
                                    .setContent(`Monto de Recargo: $${montoRecargo}`),
                                new TextDisplayBuilder()
                                    .setContent(`Monto Total: $${montoTotal}`)
                            )
                            .addSeparatorComponents(
                                new SeparatorBuilder()
                                    .setDivider(true)
                                    .setSpacing(SeparatorSpacingSize.Small)
                            )
                            .addTextDisplayComponents(
                                new TextDisplayBuilder()
                                    .setContent('-# © Todos los derechos reservados, Chicago Real Life Roleplay | ER:LC | Spanish')
                            );
                        
                        await canalMultas.send({
                            components: [containerCobro],
                            flags: MessageFlags.IsComponentsV2
                        });
                    }
                } else {
                    console.log(`[COBRO AUTOMÁTICO] Error cobrando al usuario ${multa.usuarioId}`);
                }
            } else {
                console.log(`⚠️ Usuario ${multa.usuarioId} no tiene saldo para cobro automático (tiene ${balanceUsuario}, necesita ${montoTotal})`);
            }
        }
    }
    
    await guardarMultas(multas);
    console.log('[REVISIÓN] Revisión completada.');
}
