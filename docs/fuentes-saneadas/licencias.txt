// Helpers
function buildQuestionEmbed(licenseType, question, index, total) {
  return new EmbedBuilder()
    .setTitle(`Examen teórico – Licencia ${licenseType}`)
    .setDescription(`📝 **${question.question}**\n\nSelecciona la respuesta correcta en el menú de abajo.`)
    .setFooter({ text: `Pregunta ${index + 1} de ${total}` })
    .setColor(0x00a8ff);
}

function buildSelectMenu(licenseType, userId, question) {
  return new ActionRowBuilder().addComponents(
    new StringSelectMenuBuilder()
      .setCustomId(`licencia_${licenseType}_${userId}`)
      .setPlaceholder('Selecciona una opción...')
      .setMinValues(1)
      .setMaxValues(1)
      .addOptions(
        question.options.map((opt, idx) => ({
          label: opt.length > 100 ? opt.slice(0, 97) + '...' : opt,
          value: String(idx),
        })),
      ),
  );
}

function buildResultEmbed(licenseType, correctCount, total, passed) {
  const color = passed ? 0x00ff00 : 0xff0000;
  const estado = passed ? '✅ Has APROBADO el examen.' : '❌ No alcanzaste la nota mínima para aprobar.';
  return new EmbedBuilder()
    .setTitle(`Resultado – Licencia ${licenseType}`)
    .setDescription(`Respuestas correctas: **${correctCount}/${total}**\n\n${estado}`)
    .setColor(color);
}

// Función para registrar licencia en el archivo JSON
function registrarLicencia(userId, licenseType) {
  try {
    // Leer el archivo actual
    let data = {};
    if (fs.existsSync(LICENCIAS_FILE)) {
      const fileContent = fs.readFileSync(LICENCIAS_FILE, 'utf8');
      data = fileContent ? JSON.parse(fileContent) : {};
    }
    
    // Inicializar usuario si no existe
    if (!data[userId]) {
      data[userId] = {
        userId: userId,
        licencias: []
      };
    }
    
    // Verificar si ya tiene esta licencia
    const licenciaExistente = data[userId].licencias.find(l => l.tipo === licenseType);
    
    if (!licenciaExistente) {
      // Agregar nueva licencia
      const nuevaLicencia = {
        tipo: licenseType,
        estado: 'Activa',
        fechaObtencion: new Date().toISOString(),
        fechaExpiracion: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000).toISOString() // 1 año
      };
      
      data[userId].licencias.push(nuevaLicencia);
      
      // Guardar en archivo
      fs.writeFileSync(LICENCIAS_FILE, JSON.stringify(data, null, 2));
      console.log(`✅ Licencia ${licenseType} registrada para usuario ${userId}`);
      return true;
    } else {
      // Actualizar estado si ya existe
      licenciaExistente.estado = 'Activa';
      licenciaExistente.fechaActualizacion = new Date().toISOString();
      
      fs.writeFileSync(LICENCIAS_FILE, JSON.stringify(data, null, 2));
      console.log(`✅ Licencia ${licenseType} actualizada para usuario ${userId}`);
      return true;
    }
  } catch (error) {
    console.error('❌ Error registrando licencia:', error);
    return false;
  }
}

// Función para obtener licencias de un usuario
function obtenerLicenciasUsuario(userId) {
  try {
    if (!fs.existsSync(LICENCIAS_FILE)) {
      return [];
    }
    
    const fileContent = fs.readFileSync(LICENCIAS_FILE, 'utf8');
    const data = fileContent ? JSON.parse(fileContent) : {};
    
    if (data[userId] && data[userId].licencias) {
      return data[userId].licencias;
    }
    
    return [];
  } catch (error) {
    console.error('❌ Error obteniendo licencias:', error);
    return [];
  }
}

// Función para revisar todas las licencias en el servidor
async function revisarLicenciasServidor(guild) {
  try {
    const guildId = 'ID_POR_CONFIGURAR'; // ID del servidor específico
    if (guild.id !== guildId) {
      console.log(`⚠️ Este comando solo funciona en el servidor específico: ${guildId}`);
      return { success: false, message: 'Este comando solo funciona en el servidor específico.' };
    }
    
    console.log(`🔍 Iniciando revisión de licencias en el servidor ${guild.name}...`);
    
    // Obtener todos los miembros del servidor
    await guild.members.fetch();
    const members = guild.members.cache;
    
    console.log(`📊 Total de miembros en el servidor: ${members.size}`);
    
    let data = {};
    if (fs.existsSync(LICENCIAS_FILE)) {
      const fileContent = fs.readFileSync(LICENCIAS_FILE, 'utf8');
      data = fileContent ? JSON.parse(fileContent) : {};
    }
    
    let totalUsuariosConLicencias = 0;
    let totalLicenciasRegistradas = 0;
    
    // Revisar cada miembro
    for (const [memberId, member] of members) {
      const userLicencias = [];
      
      // Verificar cada rol de licencia
      for (const [roleId, licenseType] of Object.entries(ROLE_TO_LICENSE)) {
        if (member.roles.cache.has(roleId)) {
          userLicencias.push({
            tipo: licenseType,
            estado: 'Activa',
            fechaObtencion: new Date().toISOString(),
            fechaExpiracion: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000).toISOString()
          });
        }
      }
      
      // Si tiene alguna licencia, registrar en JSON
      if (userLicencias.length > 0) {
        if (!data[memberId]) {
          data[memberId] = {
            userId: memberId,
            username: member.user.username,
            discriminator: member.user.discriminator,
            licencias: []
          };
        }
        
        // Actualizar licencias
        data[memberId].licencias = userLicencias;
        data[memberId].username = member.user.username;
        data[memberId].discriminator = member.user.discriminator;
        
        totalUsuariosConLicencias++;
        totalLicenciasRegistradas += userLicencias.length;
      } else if (data[memberId]) {
        // Si no tiene roles pero está en el archivo, marcar como inactivas
        data[memberId].licencias.forEach(licencia => {
          licencia.estado = 'Inactiva';
        });
      }
    }
    
    // Guardar los cambios
    fs.writeFileSync(LICENCIAS_FILE, JSON.stringify(data, null, 2));
    
    console.log(`✅ Revisión completada:
    - Usuarios con licencias: ${totalUsuariosConLicencias}
    - Total de licencias registradas: ${totalLicenciasRegistradas}`);
    
    return {
      success: true,
      totalUsuarios: members.size,
      usuariosConLicencias: totalUsuariosConLicencias,
      totalLicencias: totalLicenciasRegistradas
    };
    
  } catch (error) {
    console.error('❌ Error revisando licencias del servidor:', error);
    return { success: false, error: error.message };
  }
}

module.exports = {
  data: new SlashCommandBuilder()
    .setName('solicitar-licencia')
    .setDescription('Realiza el examen teórico para una licencia específica.')
    .addStringOption((option) =>
      option
        .setName('tipo')
        .setDescription('Tipo de licencia que deseas solicitar.')
        .setRequired(true)
        .addChoices(
          { name: 'A1 – Motocicletas livianas', value: 'A1' },
          { name: 'A2 – Motocicletas medianas', value: 'A2' },
          { name: 'A3 – Motocicletas alta cilindrada', value: 'A3' },
          { name: 'B1 – Vehículos livianos', value: 'B1' },
          { name: 'B2 – Microbuses / carga liviana', value: 'B2' },
          { name: 'B3 – Pesados con remolque', value: 'B3' },
          { name: 'C1 – Carga liviana', value: 'C1' },
          { name: 'C2 – Carga media', value: 'C2' },
          { name: 'C3 – Carga pesada', value: 'C3' },
          { name: 'D1 – Bus pequeño', value: 'D1' },
          { name: 'D2 – Bus mediano / grande', value: 'D2' },
          { name: 'D3 – Bus articulado / gran capacidad', value: 'D3' },
        ),
    ),

  async execute(interaction) {
    if (!interaction.guild) {
      return interaction.reply({
        content: 'Este comando solo puede usarse dentro de un servidor.',
        ephemeral: true,
      });
    }

    const licenseType = interaction.options.getString('tipo');
    const questions = licenseQuestions[licenseType];

    if (!questions) {
      return interaction.reply({
        content: 'Tipo de licencia no válido.',
        ephemeral: true,
      });
    }

    let currentIndex = 0;
    let correctCount = 0;
    const total = questions.length;

    const firstQuestion = questions[0];

    await interaction.reply({
      embeds: [buildQuestionEmbed(licenseType, firstQuestion, currentIndex, total)],
      components: [buildSelectMenu(licenseType, interaction.user.id, firstQuestion)],
      ephemeral: true,
    });

    const filter = (i) =>
      i.customId === `licencia_${licenseType}_${interaction.user.id}` &&
      i.user.id === interaction.user.id;

    const collector = interaction.channel.createMessageComponentCollector({
      filter,
      time: 15 * 60 * 1000, // 15 minutos
    });

    collector.on('collect', async (i) => {
      const question = questions[currentIndex];
      const selectedIndex = parseInt(i.values[0], 10);
      const selectedText = question.options[selectedIndex];

      if (selectedText === question.correct) {
        correctCount++;
      }

      currentIndex++;

      if (currentIndex < total) {
        const nextQuestion = questions[currentIndex];
        await i.update({
          embeds: [buildQuestionEmbed(licenseType, nextQuestion, currentIndex, total)],
          components: [buildSelectMenu(licenseType, interaction.user.id, nextQuestion)],
        });
      } else {
        const passed = "VALOR_RETIRADO";
        const resultEmbed = buildResultEmbed(licenseType, correctCount, total, passed);

        await i.update({
          embeds: [resultEmbed],
          components: [],
        });

        collector.stop('completed');

        if (passed) {
          const roleId = LICENSE_ROLES[licenseType];
          if (roleId) {
            try {
              const member = await interaction.guild.members.fetch(interaction.user.id);
              if (!member.roles.cache.has(roleId)) {
                await member.roles.add(roleId);
              }

              // Registrar la licencia en el archivo JSON
              const registroExitoso = registrarLicencia(interaction.user.id, licenseType);
              
              if (registroExitoso) {
                await interaction.followUp({
                  content: `🎉 Has aprobado la licencia **${licenseType}**. Se te ha asignado el rol <@&${roleId}> y se ha registrado en el sistema.`,
                  ephemeral: true,
                });
              } else {
                await interaction.followUp({
                  content: `🎉 Has aprobado la licencia **${licenseType}**. Se te ha asignado el rol <@&${roleId}>, pero hubo un error registrándola en el sistema.`,
                  ephemeral: true,
                });
              }
            } catch (err) {
              console.error('Error asignando rol de licencia:', err);
              await interaction.followUp({
                content: 'Aprobaste el examen, pero hubo un error al asignarte el rol. Informa a un administrador.',
                ephemeral: true,
              });
            }
          } else {
            await interaction.followUp({
              content: 'Aprobaste el examen, pero no hay un rol configurado para esta licencia. Informa a un administrador.',
              ephemeral: true,
            });
          }
        } else {
          await interaction.followUp({
            content: 'No aprobaste el examen de licencia. Puedes volver a intentarlo cuando te sientas preparado.',
            ephemeral: true,
          });
        }
      }
    });

    collector.on('end', async (collected, reason) => {
      if (reason !== 'completed') {
        try {
          await interaction.followUp({
            content: '⏰ El examen de licencia se canceló por inactividad. Usa el comando de nuevo para empezar otra vez.',
            ephemeral: true,
          });
        } catch {
          // Ignorar errores si ya no se puede responder
        }
      }
    });
  },

  // Comando para revisar licencias
  revisarLicencias: {
    data: new SlashCommandBuilder()
      .setName('revisar-licencias')
      .setDescription('Revisa todas las licencias en el servidor y actualiza la base de datos.')
      .setDefaultMemberPermissions(PermissionsBitField.Flags.Administrator),

    async execute(interaction) {
      if (!interaction.guild) {
        return interaction.reply({
          content: 'Este comando solo puede usarse dentro de un servidor.',
          ephemeral: true,
        });
      }

      // Verificar permisos de administrador
      if (!interaction.member.permissions.has(PermissionsBitField.Flags.Administrator)) {
        return interaction.reply({
          content: '❌ Solo los administradores pueden usar este comando.',
          ephemeral: true,
        });
      }

      // Enviar mensaje inicial
      await interaction.reply({
        content: '🔍 Iniciando revisión de licencias en el servidor... Esto puede tomar unos momentos.',
        ephemeral: true,
      });

      try {
        // Revisar licencias
        const resultado = await revisarLicenciasServidor(interaction.guild);
        
        if (resultado.success) {
          const embed = new EmbedBuilder()
            .setTitle('✅ Revisión de Licencias Completada')
            .setDescription('Se ha actualizado la base de datos de licencias con la información actual del servidor.')
            .addFields(
              { name: 'Total de Usuarios en el Servidor', value: resultado.totalUsuarios.toString(), inline: true },
              { name: 'Usuarios con Licencias', value: resultado.usuariosConLicencias.toString(), inline: true },
              { name: 'Total de Licencias Registradas', value: resultado.totalLicencias.toString(), inline: true }
            )
            .setColor(0x00ff00)
            .setTimestamp();

          await interaction.followUp({
            embeds: [embed],
            ephemeral: true,
          });
        } else {
          await interaction.followUp({
            content: `❌ Error al revisar licencias: ${resultado.message || resultado.error}`,
            ephemeral: true,
          });
        }
      } catch (error) {
        console.error('Error ejecutando comando revisar-licencias:', error);
        await interaction.followUp({
          content: `❌ Error inesperado: ${error.message}`,
          ephemeral: true,
        });
      }
    }
  }
};
