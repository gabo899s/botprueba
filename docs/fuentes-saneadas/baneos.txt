async function leerBaneados() {
    try {
        await fs.access(BANEADOS_FILE);
        const data = await fs.readFile(BANEADOS_FILE, 'utf8');
        return JSON.parse(data);
    } catch (error) {
        return {};
    }
}

async function guardarBaneados(data) {
    try {
        await fs.mkdir(path.dirname(BANEADOS_FILE), { recursive: true });
        await fs.writeFile(BANEADOS_FILE, JSON.stringify(data, null, 4), 'utf8');
    } catch (error) {
        console.error('Error guardando baneados:', error);
    }
}

function calcularTiempoBan(tiempo) {
    // Expresión regular para formato: número + unidad (s, m, h, d, w, M)
    const regex = /^(\d+)([smhdwM])$/;
    const match = tiempo.match(regex);
    
    if (!match) {
        throw new Error('Formato de tiempo inválido. Usa: 1s, 1m, 1h, 1d, 1w, 1M\nEjemplos: 30s, 45m, 2h, 3d, 1w, 2M');
    }
    
    const cantidad = parseInt(match[1]);
    const unidad = match[2];
    
    const unidades = {
        's': 1000,           // segundos
        'm': 60000,          // minutos
        'h': 3600000,        // horas
        'd': 86400000,       // días
        'w': 604800000,      // semanas (7 días)
        'M': 2592000000      // meses (30 días aprox)
    };
    
    if (!unidades[unidad]) {
        throw new Error(`Unidad de tiempo no válida: ${unidad}. Usa: s, m, h, d, w, M`);
    }
    
    return cantidad * unidades[unidad];
}

async function obtenerIdRoblox(nombreUsuario) {
    try {
        console.log(`🔍 Buscando usuario de Roblox: ${nombreUsuario}`);
        
        // Primero intentamos obtener por nombre de usuario
        const response = await axios.post('https://users.roblox.com/v1/usernames/users', {
            usernames: [nombreUsuario],
            excludeBannedUsers: false
        }, {
            timeout: 10000,
            headers: {
                'Content-Type': 'application/json',
                'Accept': 'application/json'
            }
        });
        
        if (response.data.data && response.data.data.length > 0) {
            const usuario = response.data.data[0];
            console.log(`✅ Usuario encontrado: ${usuario.name} (ID: ${usuario.id})`);
            return {
                id: usuario.id,
                name: usuario.name,
                displayName: usuario.displayName || usuario.name
            };
        }
        
        // Si no se encuentra por nombre, verificar si el input es un ID numérico
        const idNumero = parseInt(nombreUsuario);
        if (!isNaN(idNumero)) {
            try {
                const responseById = await axios.get(`https://users.roblox.com/v1/users/${idNumero}`, {
                    timeout: 10000
                });
                
                if (responseById.data) {
                    console.log(`✅ ID de Roblox encontrado: ${responseById.data.name} (ID: ${responseById.data.id})`);
                    return {
                        id: responseById.data.id,
                        name: responseById.data.name,
                        displayName: responseById.data.displayName || responseById.data.name
                    };
                }
            } catch (error) {
                // Si falla la búsqueda por ID, continuamos
            }
        }
        
        throw new Error(`Usuario de Roblox no encontrado: "${nombreUsuario}"`);
        
    } catch (error) {
        console.error('❌ Error en API de Roblox:', error.message);
        
        if (error.code === 'ECONNREFUSED') {
            throw new Error('No se puede conectar con la API de Roblox');
        }
        
        if (error.response?.status === 404) {
            throw new Error(`Usuario de Roblox no encontrado: "${nombreUsuario}"`);
        }
        
        if (error.response?.status === 400) {
            throw new Error(`Nombre de usuario inválido: "${nombreUsuario}"`);
        }
        
        throw new Error(`Error al verificar usuario de Roblox: ${error.message}`);
    }
}

function extraerUsuarioRoblox(nickname) {
    if (!nickname) return '';
    const partes = nickname.split('|');
    if (partes.length > 1) {
        return partes.slice(1).join('|').trim();
    }
    return nickname.trim();
}

async function banearEnERLC(robloxId, motivo = 'Baneado desde Discord') {
    try {
        // Usamos el ID numérico de Roblox para el comando
        await prcClient.enviarComando(`:ban ${robloxId} ${motivo}`);
        return { success: true, message: '✅ Baneo en ER:LC ejecutado' };
    } catch (error) {
        console.error('Error banear ER:LC:', error.message);
        
        // Verificar si el error es por servidor offline
        if (error.message.includes('offline') || error.message.includes('no disponible')) {
            return { 
                success: false, 
                message: '⚠️ Servidor ER:LC offline - El baneo se aplicó en Discord, pero no en ER:LC (se reintentará automáticamente)' 
            };
        }
        
        return { 
            success: false, 
            message: `❌ Error al banear en ER:LC: ${error.message}` 
        };
    }
}

async function verificarServidorActivo() {

async function revisarBaneosExpirados(client) {
    console.log('🔄 Revisando baneos expirados...');
    
    try {
        const baneados = await leerBaneados();
        const ahora = Date.now();
        let cambios = false;
        
        for (const [userId, baneo] of Object.entries(baneados)) {
            if (baneo.estado === 'activo' && ahora >= baneo.horaDesbaneo) {
                console.log(`⏰ Baneo expirado para ${userId} (${baneo.robloxName || baneo.robloxId})`);
                
                // Marcar como expirado (lo desbanearemos cuando haya jugadores)
                baneados[userId].estado = 'expirado (pendiente)';
                baneados[userId].intentosERLC = (baneados[userId].intentosERLC || 0) + 1;
                cambios = true;
            }
        }
        
        if (cambios) await guardarBaneados(baneados);
        console.log('✅ Revisión de baneos expirados completada');
        
    } catch (error) {
        console.error('❌ Error revisando baneos expirados:', error.message);
    }
}

async function procesarBaneosPendientes(client) {
    console.log('🔄 Procesando baneos pendientes...');
    
    try {
        const servidorActivo = await verificarServidorActivo();
        if (!servidorActivo) {
            console.log('⏳ Servidor ER:LC vacío/offline, esperando...');
            return;
        }
        
        const baneados = await leerBaneados();
        let cambios = false;
        
        for (const [userId, baneo] of Object.entries(baneados)) {
            // Desbanear usuarios expirados
            if (baneo.estado === 'expirado (pendiente)') {
                console.log(`🔄 Desbaneando automáticamente: ${userId} (${baneo.robloxName || baneo.robloxId})`);
                
                try {
                    // Buscar guild
                    const guild = client.guilds.cache.first();
                    if (!guild) continue;
                    
                    // Obtener miembro
                    const miembro = await guild.members.fetch(userId).catch(() => null);
                    if (miembro) {
                        // Restaurar roles
                        if (baneo.rolesAnteriores && baneo.rolesAnteriores.length > 0) {
                            await miembro.roles.set(baneo.rolesAnteriores);
                        }
                        
                        // Restaurar apodo
                        if (baneo.apodoAnterior) {
                            await miembro.setNickname(baneo.apodoAnterior);
                        }
                    }
                    
                    // Desbanear en ER:LC
                    const resultadoERLC = await desbanearEnERLC(baneo.robloxId);
                    
                    // Actualizar estado en JSON
                    baneados[userId] = {
                        ...baneo,
                        estado: 'terminado',
                        horaDesbaneoReal: Date.now(),
                        desbaneoAutomatico: true,
                        mensajeERLC: resultadoERLC.message
                    };
                    cambios = true;
                    
                    console.log(`✅ Usuario ${userId} desbaneado automáticamente`);
                    
                } catch (error) {
                    console.error(`❌ Error desbaneando ${userId}:`, error.message);
                    baneados[userId].intentosERLC = (baneados[userId].intentosERLC || 0) + 1;
                    
                    if (baneados[userId].intentosERLC >= 5) {
                        baneados[userId].estado = 'error - max intentos';
                        baneados[userId].error = `Máximo de intentos alcanzado: ${error.message}`;
                    }
                    cambios = true;
                }
            }
        }
        
        if (cambios) await guardarBaneados(baneados);
        
    } catch (error) {
        console.error('❌ Error procesando baneos pendientes:', error.message);
    }
}

        }

// Función auxiliar para formatear tiempo
function formatTiempo(segundos) {
    if (segundos < 60) return `${segundos} segundos`;
    
    const minutos = Math.floor(segundos / 60);
    if (minutos < 60) return `${minutos} minutos`;
    
    const horas = Math.floor(minutos / 60);
    if (horas < 24) return `${horas} horas`;
    
    const dias = Math.floor(horas / 24);
    return `${dias} días`;
}
