// ================== PERSISTENCIA OPTIMIZADA ==================
function cargarEncuesta() {
  if (!fs.existsSync(DATA_FILE)) {
    return null;
  }
  try {
    const data = JSON.parse(fs.readFileSync(DATA_FILE, 'utf8'));
    if (data) {
      data.votosCiviles = new Set(data.votosCiviles || []);
      data.votosMods = new Set(data.votosMods || []);
    }
    return data;
  } catch (err) {
    console.error('❌ Error cargando encuesta:', err);
    return null;
  }
}

function guardarEncuestaAsync() {
  if (!ultimaEncuesta) {
    fs.writeFile(DATA_FILE, JSON.stringify(null), (err) => {
      if (err) console.error('Error borrando archivo data:', err);
    });
    return;
  }

  const dataToSave = {
    ...ultimaEncuesta,
    votosCiviles: Array.from(ultimaEncuesta.votosCiviles),
    votosMods: Array.from(ultimaEncuesta.votosMods),
  };

  fs.writeFile(DATA_FILE, JSON.stringify(dataToSave, null, 2), (err) => {
    if (err) console.error('❌ Error guardando encuesta en background:', err);
  });
}

// ================== INTEGRACIÓN CON SHIFTS ==================
async function verificarShiftActivo(userId) {
  try {
    // Intentar cargar el módulo de shifts dinámicamente
    const shiftsModule = require('./shifts');
    if (!shiftsModule || !shiftsModule.shiftsActivos) return false;
    
    const shiftsActivos = shiftsModule.shiftsActivos();
    return shiftsActivos && shiftsActivos.has(userId);
  } catch (error) {
    // Si no existe el módulo, retornar false
    return false;
  }
}

async function cerrarShiftsPorEstadoServidor(client, estado) {
  if (estado === 'cerrado') {
    try {
      const shiftsModule = require('./shifts');
      if (shiftsModule && typeof shiftsModule.cerrarTodosShiftsPorApertura === 'function') {
        console.log('🔄 Cerrando todos los shifts por cambio de estado del servidor...');
        await shiftsModule.cerrarTodosShiftsPorApertura(client);
        
        // Crear embed de notificación
        const embed = new EmbedBuilder()
          .setColor(0xff0000)
          .setTitle('⏹️ Shifts Finalizados')
          .setDescription('Todos los shifts han sido finalizados automáticamente debido al cierre del servidor.')
          .addFields(
            { name: 'Motivo', value: 'Servidor cerrado', inline: true },
            { name: 'Acción', value: 'Shifts finalizados', inline: true },
            { name: 'Fecha', value: `<t:${Math.floor(Date.now()/1000)}:F>`, inline: true }
          )
          .setFooter({ text: 'Sistema de Aperturas - Escazú RP' });
        
        // Enviar notificación al canal de logs si existe
        const logChannel = client.channels.cache.get(APERTURAS_LOG_CHANNEL_ID);
        if (logChannel) {
          await logChannel.send({ embeds: [embed] }).catch(() => {});
        }
        
        return true;
      }
    } catch (error) {
      console.error('❌ Error cerrando shifts desde aperturas:', error);
    }
  }
  return false;
}

// ================== LOGICA DE DEBOUNCE VISUAL ==================
async function programarActualizacionVisual(client) {
  if (updateTimer) {
    pendingUpdate = true;
    return;
  }

  const ejecutarEdicion = async () => {
    if (!ultimaEncuesta) return;

    try {
      const channel = client.channels.cache.get(ultimaEncuesta.channelId);
      if (!channel) return;
      
      const message = await channel.messages.fetch(ultimaEncuesta.messageId).catch(() => null);
      if (!message) return;

      const nuevoContainer = crearContainerApertura({
        organizador: `<@${ultimaEncuesta.organizadorId}>`,
        hora: ultimaEncuesta.hora,
        minCiviles: ultimaEncuesta.minCiviles,
        minMods: ultimaEncuesta.minMods,
        votosCiviles: ultimaEncuesta.votosCiviles.size,
        votosMods: ultimaEncuesta.votosMods.size,
        incluirBotones: true,
      });

      await message.edit({
        components: [nuevoContainer],
        flags: MessageFlags.IsComponentsV2,
      });
    } catch (err) {
      console.error('❌ Error en actualización visual (debounce):', err.message);
    }
  };

  updateTimer = setTimeout(async () => {
    updateTimer = null;
    
    if (pendingUpdate) {
      pendingUpdate = false;
      await ejecutarEdicion();
    }
  }, 2000);

  await ejecutarEdicion();
}

// ================== MANEJADOR DE INTERACCIONES ==================
async function handleComponentInteraction(interaction) {
  if (!interaction.isButton()) return false;

  const customId = interaction.customId;

  if (customId.startsWith('apertura_')) {
    await handleAperturaInteraction(interaction);
    return true;
  }
  return false;
}

async function handleAperturaInteraction(interaction) {
  // 1. Anti-Spam Check
  const userId = interaction.user.id;
  const now = Date.now();
  if (userCooldowns.has(userId)) {
    const expirationTime = userCooldowns.get(userId) + 2000;
    if (now < expirationTime) {
      return interaction.reply({
        content: '⏳ Por favor, espera unos segundos antes de votar de nuevo.',
        ephemeral: true,
        flags: MessageFlags.Ephemeral,
      });
    }
  }
  userCooldowns.set(userId, now);

  // 2. Validación de Estado
  if (!ultimaEncuesta) {
    return interaction.reply({
      components: [crearContainerMensaje({ tipo: 'error', titulo: 'Encuesta no encontrada', descripcion: 'No hay una encuesta activa.' })],
      flags: MessageFlags.Ephemeral | MessageFlags.IsComponentsV2,
    });
  }

  // 3. Verificar si el usuario tiene un shift activo (INTEGRACIÓN CON SHIFTS)
  const tieneShiftActivo = await verificarShiftActivo(userId);
  if (tieneShiftActivo) {
    return interaction.reply({
      components: [crearContainerMensaje({ 
        tipo: 'error', 
        titulo: 'Shift activo detectado',
        descripcion: 'Tienes un shift activo en el sistema de moderación.\n\n⏹️ **Termina tu shift antes de votar en aperturas.**\n\n📍 Ve al canal de shifts y usa el botón "Terminar Shift" o el comando correspondiente.' 
      })],
      flags: MessageFlags.Ephemeral | MessageFlags.IsComponentsV2,
    });
  }

  // 4. Proceso de Voto (EN RAM - INSTANTÁNEO)
  const customId = interaction.customId;
  const votosCiviles = ultimaEncuesta.votosCiviles;
  const votosMods = ultimaEncuesta.votosMods;
  let replyMsg = null;
  let logText = null;

  if (customId === 'apertura_civil') {
    const eraMod = votosMods.has(userId);
    votosMods.delete(userId);
    votosCiviles.add(userId);
    replyMsg = { tipo: 'success', titulo: 'Voto registrado', descripcion: 'Votaste como **Civil listo**.' };
    logText = eraMod ? `🟢 <@${userId}> cambió a **Civil**.` : `🟢 <@${userId}> votó **Civil**.`;

  } else if (customId === 'apertura_mod') {
    // Check de rol
    const member = interaction.member;
    const isStaff = member && member.roles && member.roles.cache && member.roles.cache.has(STAFF_ROLE_ID);
    if (!isStaff) {
      return interaction.reply({
        components: [crearContainerMensaje({ tipo: 'error', titulo: 'Solo staff', descripcion: 'Solo Staff puede votar Moderador.' })],
        flags: MessageFlags.Ephemeral | MessageFlags.IsComponentsV2,
      });
    }
    const eraCivil = votosCiviles.has(userId);
    votosCiviles.delete(userId);
    votosMods.add(userId);
    replyMsg = { tipo: 'success', titulo: 'Voto registrado', descripcion: 'Votaste como **Moderador listo**.' };
    logText = eraCivil ? `🛡️ <@${userId}> cambió a **Moderador**.` : `🛡️ <@${userId}> votó **Moderador**.`;

  } else if (customId === 'apertura_quitar') {
    votosCiviles.delete(userId);
    votosMods.delete(userId);
    replyMsg = { tipo: 'info', titulo: 'Voto retirado', descripcion: 'Has retirado tu voto.' };
    logText = `🗑️ <@${userId}> retiró su voto.`;
  }

  // 5. Guardado en Disco (Background)
  guardarEncuestaAsync();

  // 6. Respuesta al Usuario (Inmediata)
  await interaction.reply({
    components: [crearContainerMensaje(replyMsg)],
    flags: MessageFlags.Ephemeral | MessageFlags.IsComponentsV2,
  });

  // 7. Actualización Visual (Debounced / Throttled)
  programarActualizacionVisual(interaction.client);

  // 8. Logs
  const logChannel = interaction.client.channels.cache.get(APERTURAS_LOG_CHANNEL_ID);
  if (logChannel) {
    logChannel.send({
      components: [crearContainerMensaje({ tipo: 'info', titulo: 'Registro de voto', descripcion: logText })],
      flags: MessageFlags.IsComponentsV2,
    }).catch(console.error);
  }
}

async function manejarEncuestaApertura(interaction) {
  const minCiviles = interaction.options.getInteger('min-votos', true);
  const minMods = interaction.options.getInteger('min-mods', true);
  const hora = interaction.options.getString('hora', true);
  const organizador = interaction.user;

  // ==== LIMPIEZA DE ENCUESTA ANTERIOR ====
  if (ultimaEncuesta) {
    try {
      const oldChannel = interaction.client.channels.cache.get(ultimaEncuesta.channelId);
      if (oldChannel) {
        const oldMsg = await oldChannel.messages.fetch(ultimaEncuesta.messageId).catch(() => null);
        if (oldMsg) {
            const containerDesactivado = crearContainerApertura({
                organizador: `<@${ultimaEncuesta.organizadorId}>`,
                hora: ultimaEncuesta.hora,
                minCiviles: ultimaEncuesta.minCiviles,
                minMods: ultimaEncuesta.minMods,
                votosCiviles: ultimaEncuesta.votosCiviles.size,
                votosMods: ultimaEncuesta.votosMods.size,
                incluirBotones: true,
                botonesDeshabilitados: true
            });
            
            containerDesactivado.addTextDisplayComponents(t => t.setContent('🔴 **Esta encuesta ha finalizado.**'));
            
            await oldMsg.edit({ components: [containerDesactivado], flags: MessageFlags.IsComponentsV2 });
        }
      }
    } catch (err) {
      console.error('Error cerrando encuesta anterior:', err);
    }
  }

  // ==== CREAR NUEVA ENCUESTA ====
  const votosCiviles = new Set();
  const votosMods = new Set();
  
  const container = crearContainerApertura({
    organizador: `<@${organizador.id}>`,
    hora,
    minCiviles,
    minMods,
    votosCiviles: 0,
    votosMods: 0,
    incluirBotones: true,
  });

  await interaction.reply({ components: [container], flags: MessageFlags.IsComponentsV2 });
  const message = await interaction.fetchReply();

  ultimaEncuesta = {
    organizadorId: organizador.id,
    hora,
    minCiviles,
    minMods,
    votosCiviles,
    votosMods,
    messageId: message.id,
    channelId: message.channel.id,
    guildId: message.guild.id,
  };

  guardarEncuestaAsync();

  const logChannel = interaction.client.channels.cache.get(APERTURAS_LOG_CHANNEL_ID);
  if (logChannel) {
    const containerLog = crearContainerApertura({
      organizador: `<@${organizador.id}>`,
      hora,
      minCiviles,
      minMods,
      votosCiviles: 0,
      votosMods: 0,
      incluirBotones: false,
    });
    await logChannel.send({ components: [containerLog], flags: MessageFlags.IsComponentsV2 }).catch(() => {});
  }
}

async function manejarEstadoServidor(interaction) {
    const estado = interaction.options.getString('estado', true);
    const organizador = interaction.user;
    const ahora = Math.floor(Date.now() / 1000);
  
    const containerEstado = crearContainerEstadoServidor({
      estado,
      organizadorId: organizador.id,
      ahora,
    });
  
    await interaction.reply({
      components: [containerEstado],
      flags: MessageFlags.IsComponentsV2,
    });
  
    const logChannel = interaction.client.channels.cache.get(APERTURAS_LOG_CHANNEL_ID);
  
    // Enviar al log
    if (logChannel) {
      const containerLog = crearContainerEstadoServidor({
        estado,
        organizadorId: organizador.id,
        ahora,
      });
  
      await logChannel.send({
        components: [containerLog],
        flags: MessageFlags.IsComponentsV2,
      });
    }
  
    // ================== INTEGRACIÓN CON SHIFTS ==================
    // Cerrar todos los shifts cuando el servidor se cierra
    const shiftsCerrados = await cerrarShiftsPorEstadoServidor(interaction.client, estado);
    
    // Si hay encuesta activa, manejar DMs y resumen
    if (ultimaEncuesta) {
      const civIds = Array.from(ultimaEncuesta.votosCiviles || []);
      const modIds = Array.from(ultimaEncuesta.votosMods || []);
  
      if (estado === 'abierto') {
        const todosIds = Array.from(new Set([...civIds, ...modIds]));
  
        for (const id of todosIds) {
          interaction.client.users.fetch(id).then(user => {
            const dmContainer = crearContainerDmApertura(ultimaEncuesta, estado);
            user.send({
                components: [dmContainer],
                flags: MessageFlags.IsComponentsV2,
            }).catch(() => {});
          }).catch(() => {});
        }
      }
  
      if (logChannel) {
        const resumenContainer = crearContainerResumenApertura(
          ultimaEncuesta,
          estado,
          civIds,
          modIds,
        );
  
        await logChannel.send({
          components: [resumenContainer],
          flags: MessageFlags.IsComponentsV2,
        });
      }
  
      // Limpiar la encuesta después de cambiar el estado
      ultimaEncuesta = null;
      guardarEncuestaAsync();
    }
  }

// ================== EXPORTS ==================
