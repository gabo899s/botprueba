    function showToast(msg, type = "info") {
        const container = document.getElementById('toast-container');

        let icon = type === 'error' ? 'circle-xmark' : type === 'success' ? 'circle-check' : 'circle-info';

    // Auth Dev
    async function submitDevKey() {
      const key = document.getElementById('dev-key-input').value;
      if (!key) { showToast('Ingrese una clave', 'error'); return; }
      try {
        const res = await fetch('/api/dev/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ clave: key }) });
        const data = await res.json();
        if (data.success) {
          localStorage.setItem('judicial_token', data.token);
          window.location.href = '/v2/police/dashboard';
        } else showToast('Clave inválida', 'error');
      } catch (e) { showToast('Error de conexión', 'error'); }
    }

    // Routing
    async function handleRouting() {
        const path = window.location.pathname;
        const urlParams = new URLSearchParams(window.location.search);
        const token = "VALOR_RETIRADO";
        const error = urlParams.get('error');

        if(error === 'no_permission') showToast('No tiene permisos para acceder.', 'error');
        if(token) { localStorage.setItem('judicial_token', token); window.history.replaceState({}, '', '/v2/police/dashboard'); }

        const storedToken = "VALOR_RETIRADO";
        
        // Si estamos en /v2/police/dashboard y hay token, mostrar dashboard
        if (path === '/v2/police/dashboard' && storedToken) {
            try {
                const res = await fetch('/api/auth/me', { headers: { 'Authorization': `Bearer ${storedToken}` } });
                if (res.ok) {
                    const data = await res.json();
                    if (data.success) {
                        currentUser = data.user;
                        document.getElementById('view-landing').style.display = 'none';
                        document.getElementById('view-dashboard').classList.remove('hidden-view');
                        document.body.classList.remove('bg-gradient');
                        lucide.createIcons();
                        updateUserInfo();
                        return;
                    }
                }
                throw new Error('Sesión inválida');
            } catch (e) { 
                localStorage.removeItem('judicial_token'); 
                document.getElementById('view-landing').style.display = 'block'; 
                document.getElementById('view-dashboard').classList.add('hidden-view');
            }
        } 
        // Si estamos en /v2/police (sin dashboard) mostrar landing de plataforma
        else if (path === '/v2/police') {
            document.getElementById('view-landing').style.display = 'block';
            document.getElementById('view-dashboard').classList.add('hidden-view');
        }
        // Cualquier otra ruta bajo /v2/police redirige a dashboard si hay token
        else if (path.startsWith('/v2/police/')) {
            if (storedToken) {
                window.location.href = '/v2/police/dashboard';
            } else {
                window.location.href = '/v2/police';
            }
        }
    }

    // Search
    document.getElementById('search-input')?.addEventListener('input', function(e) {
        const query = e.target.value.trim();
        if (searchTimeout) clearTimeout(searchTimeout);
        if (!query || query.length < 2) { document.getElementById('search-dropdown').classList.remove('active'); return; }
        searchTimeout = setTimeout(async () => {
            try {
                const res = await fetch(`/api/search/suggestions?q=${encodeURIComponent(query)}`);
                const data = await res.json();
                const dd = document.getElementById('search-dropdown');
                dd.innerHTML = '';
                if (data.success && data.suggestions.length > 0) {
                    data.suggestions.forEach(r => {
                        dd.innerHTML += `<div class="search-result" onclick='selectSearchResult(${JSON.stringify(r)})'><img src="${r.robloxAvatar}" class="w-8 h-8 rounded-full"><div><div class="font-bold text-sm text-white">${r.nombreCompleto}</div><div class="text-xs text-gray-400">${r.cedula}</div></div></div>`;
                    });
                    dd.classList.add('active');
                } else dd.classList.remove('active');
            } catch (e) {}
        }, 300);
    });

    async function selectSearchResult(r) {
        document.getElementById('search-dropdown').classList.remove('active');
        document.getElementById('search-input').value = r.cedula;
        const res = await fetch(`/api/search?q=${encodeURIComponent(r.cedula)}`);
        const data = await res.json();
        if(data.success) { selectedCitizen = data.data; renderCitizen(data.data); }
    }

        document.getElementById('dashboard-empty').classList.add('hidden-view');

        lucide.createIcons();
    }

    async function submitMulta() {
        const token = "VALOR_RETIRADO";
        if (!token) {
            showToast('Debe iniciar sesión primero', 'error');
            window.location.href = '/v2/police';
            return;
        }
        
        try {
            const res = await fetch('/api/multa', {
                method: 'POST', 
                headers: {
                    'Content-Type': 'application/json', 
                    'Authorization': `Bearer ${token}`
                },
                body: JSON.stringify({
                    oficial: currentUser.discordId, 
                    ciudadano: `${selectedCitizen.nombre} ${selectedCitizen.apellidos}`, 
                    documento: selectedCitizen.numeroCedula,
                    placa: document.getElementById('multa-placa').value, 
                    valor: document.getElementById('multa-valor').value, 
                    articulos: document.getElementById('multa-articulos').value,
                    oficialDiscordId: currentUser.discordId, 
                    oficialNombre: currentUser.nombreCompleto
                })
            });
            const data = await res.json();
            if(data.success) { 
                showToast('Multa generada exitosamente', 'success'); 
                window.open(data.pdfUrl, '_blank'); 
                closeModal('multa'); 
                selectSearchResult(selectedCitizen); 
            } else {
                showToast(data.message || 'Error al generar multa', 'error');
            }
        } catch(e) { 
            showToast('Error de conexión con el servidor', 'error'); 
        }
    }

    async function submitArresto() {
        const token = "VALOR_RETIRADO";
        if (!token) {
            showToast('Debe iniciar sesión primero', 'error');
            window.location.href = '/v2/police';
            return;
        }
        
        const fd = new FormData();
        fd.append('foto', document.getElementById('arresto-foto').files[0]);
        fd.append('oficial', currentUser.discordId); 
        fd.append('ciudadano', `${selectedCitizen.nombre} ${selectedCitizen.apellidos}`);
        fd.append('documento', selectedCitizen.numeroCedula); 
        fd.append('tiempo', document.getElementById('arresto-tiempo').value);
        fd.append('articulos', document.getElementById('arresto-articulos').value); 
        fd.append('oficialDiscordId', currentUser.discordId); 
        fd.append('oficialNombre', currentUser.nombreCompleto);
        
        try {
            const res = await fetch('/api/arresto', { 
                method: 'POST', 
                headers: {'Authorization': `Bearer ${token}`}, 
                body: fd 
            });
            const data = await res.json();
            if(data.success) { 
                showToast('Arresto procesado exitosamente', 'success'); 
                window.open(data.pdfUrl, '_blank'); 
                closeModal('arresto'); 
                selectSearchResult(selectedCitizen); 
            } else {
                showToast(data.message || 'Error al procesar arresto', 'error');
            }
        } catch(e) { 
            showToast('Error de conexión con el servidor', 'error'); 
        }
    }
