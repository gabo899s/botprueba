        // Funciones de autenticación
        async function checkAuth() {
            const token = "VALOR_RETIRADO";
            
            if (!token) {
                renderUnauthenticatedUI();
                return;
            }

            try {
                const response = await fetch('/api/auth/me', {
                    headers: {
                        'Authorization': `Bearer ${token}`
                    }
                });

                if (!response.ok) {
                    if (response.status === 401 || response.status === 403) {
                        localStorage.removeItem('judicial_token');
                        renderUnauthenticatedUI();
                        return;
                    }
                    throw new Error('Error en la respuesta del servidor');
                }

                const data = await response.json();
                
                if (data.success && data.user) {
                    currentUser = data.user;
                    renderAuthenticatedUI();
                    
                    // Mostrar modal de Roblox si no está vinculado y no se ha mostrado en esta sesión
                    if (!currentUser.robloxLinked && !robloxModalShown) {
                        setTimeout(() => {
                            showRobloxModal();
                        }, 1000);
                    }
                } else {
                    localStorage.removeItem('judicial_token');
                    renderUnauthenticatedUI();
                }
            } catch (error) {
                console.error('Error checking auth:', error);
                localStorage.removeItem('judicial_token');
                renderUnauthenticatedUI();
            }
        }

            const connectionsList = dropdown.querySelector('#connections-list');
            
            if (currentUser.roblox && currentUser.roblox.username) {
                // Tiene cuenta de Roblox vinculada
                const robloxTemplate = document.getElementById('roblox-connection-template');
                const robloxConnection = robloxTemplate.content.cloneNode(true).querySelector('.connection-item');
                
                robloxConnection.querySelector('.connection-avatar').src = currentUser.roblox.avatarUrl || 'https://tr.rbxcdn.com/56885563943362145719904229562725/150/150/AvatarHeadshot/Png';
                robloxConnection.querySelector('.connection-username').textContent = currentUser.roblox.username;
                
                connectionsList.appendChild(robloxConnection);
            } else {
                // No tiene Roblox vinculado
                const connectTemplate = document.getElementById('connect-roblox-template');
                const connectBtn = connectTemplate.content.cloneNode(true);
                connectionsList.appendChild(connectBtn);
            }

            // Agregar al DOM
            document.getElementById('user-area').appendChild(dropdown);
            profileDropdown = dropdown;

            // Cerrar dropdown al hacer clic fuera
            document.addEventListener('click', closeDropdownOnClickOutside);
        }

        // Alternar visibilidad del dropdown
        function toggleProfileDropdown(e) {
            e.stopPropagation();
            
            const trigger = document.getElementById('profile-trigger');
            if (!trigger || !profileDropdown) return;

            trigger.classList.toggle('active');
            profileDropdown.classList.toggle('show');
        }

        // Cerrar dropdown al hacer clic fuera
        function closeDropdownOnClickOutside(e) {
            const trigger = document.getElementById('profile-trigger');
            if (!trigger || !profileDropdown) return;

            if (!trigger.contains(e.target) && !profileDropdown.contains(e.target)) {
                trigger.classList.remove('active');
                profileDropdown.classList.remove('show');
            }
        }

        // Conectar con Roblox
        function connectRoblox() {
            const token = "VALOR_RETIRADO";
            if (!token) {
                showToast('Debes iniciar sesión primero', 'error');
                return;
            }
            
            // Redirigir a la autenticación de Roblox con el token de Discord
            window.location.href = `/auth/roblox?discord_token="VALOR_RETIRADO";
        }

        // Iniciar login con Roblox desde modal
        function initiateRobloxLogin() {
            closeRobloxModal();
            connectRoblox();
        }

        // Actualizar información de Roblox
        function updateRobloxInfo() {

            const trigger = document.getElementById('profile-trigger');
            if (trigger && profileDropdown) {
                trigger.classList.remove('active');
                profileDropdown.classList.remove('show');
            }
        }

        // Mostrar mensaje de "próximamente"
        function showComingSoon() {

        }

        // Cerrar sesión
        function logout() {
            localStorage.removeItem('judicial_token');
            sessionStorage.removeItem('robloxModalShown');
            currentUser = null;
            
            // Cerrar dropdown si está abierto
            const trigger = document.getElementById('profile-trigger');
            if (trigger && profileDropdown) {
                trigger.classList.remove('active');
                profileDropdown.classList.remove('show');
            }
            
            // Remover event listener
            document.removeEventListener('click', closeDropdownOnClickOutside);
            
            // Recargar UI
            renderUnauthenticatedUI();
            showToast('Sesión cerrada correctamente', 'success');
            
            // Limpiar parámetros de URL
            window.history.replaceState({}, document.title, '/');
        }

        // Manejar token de URL (si viene del callback de Discord)
        function handleUrlToken() {
            const urlParams = new URLSearchParams(window.location.search);
            const token = "VALOR_RETIRADO";
            const message = urlParams.get('message');
            const redirect = urlParams.get('redirect');
            const error = urlParams.get('error');
            
            if (token) {
                localStorage.setItem('judicial_token', token);
                
                // Mostrar mensaje apropiado
                if (message === 'discord_logged_in') {
                    showToast('¡Sesión iniciada con Discord correctamente!', 'success');
                }
                
                // Limpiar URL
                window.history.replaceState({}, document.title, '/');
                
                // Recargar autenticación
                checkAuth();
            } else if (redirect === 'policia' && message === 'login_required') {
                showToast('Debes iniciar sesión para acceder a la Plataforma Policial', 'info');
                // Limpiar URL
                window.history.replaceState({}, document.title, '/');
            } else if (error) {
                const errorMessages = {
                    'no_code': 'No se recibió código de autorización',
                    'auth_failed': 'Error al autenticar. Por favor intenta nuevamente.',
                    'no_permission': 'No tienes permisos para acceder'
                };
                const errorMsg = urlParams.get('message') || errorMessages[error] || 'Ocurrió un error';
                showToast(decodeURIComponent(errorMsg), 'error');
                // Limpiar URL
                window.history.replaceState({}, document.title, '/');
            }
        }

        // Manejar clic en Plataforma Policial
        function handlePoliciaClick(e) {
            e.preventDefault();
            
            const token = "VALOR_RETIRADO";
            
            if (!token) {
                showToast('Debes iniciar sesión con Discord para acceder a la Plataforma Policial', 'info');
                return;
            }
            
            // Si tiene token, verificar si tiene acceso policial
            if (currentUser && currentUser.hasPoliceAccess) {
                window.location.href = '/v2/police/dashboard?token="VALOR_RETIRADO";
            } else {
                showToast('No tienes los roles necesarios para acceder a la Plataforma Policial', 'error');
            }
        }

        // Manejar clic en Panel Ciudadano
        function handleCiudadanoClick(e) {
            e.preventDefault();
            
            if (!currentUser) {
                showToast('Debes iniciar sesión primero', 'error');
                return;
            }
            
            // Verificar si tiene Roblox vinculado
            if (!currentUser.robloxLinked) {
                showRobloxModal();
                return;
            }
            
            // Si tiene Roblox vinculado, ir al panel
            const token = "VALOR_RETIRADO";
            window.location.href = '/ciudadano?token="VALOR_RETIRADO";
        }

        // Mostrar modal de Roblox
        function showRobloxModal() {
            document.getElementById('roblox-modal').classList.add('show');
            sessionStorage.setItem('robloxModalShown', 'true');
            robloxModalShown = true;
        }

        // Cerrar modal de Roblox
        function closeRobloxModal() {
            document.getElementById('roblox-modal').classList.remove('show');
        }

        // Cerrar modal al hacer clic fuera
        document.getElementById('roblox-modal')?.addEventListener('click', function(e) {
            if (e.target === this) {
                closeRobloxModal();
            }
        });

        // Inicialización
        document.addEventListener('DOMContentLoaded', function() {
            handleUrlToken();
            checkAuth();
        });
