// ========== FUNCIONES AUXILIARES ==========
async function leerMultas() {
    try {
        const data = await fs.readFile(MULTAS_FILE, 'utf8');
        return JSON.parse(data);
    } catch (error) {
        console.error('Error leyendo multas:', error);
        return [];
    }
}

async function guardarMultas(multas) {
    try {
        await fs.writeFile(MULTAS_FILE, JSON.stringify(multas, null, 2), 'utf8');
        return true;
    } catch (error) {
        console.error('Error guardando multas:', error);
        return false;
    }
}

async function obtenerNombreUsuario(client, userId) {
    try {
        const usuario = await client.users.fetch(userId);
        return usuario.username;
    } catch (error) {
        console.error(`Error obteniendo usuario ${userId}:`, error);
        return 'Desconocido';
    }
}

function generarIdMulta() {
    const chars = 'abcdefghijklmnopqrstuvwxyz0123456789';
    let id = '';
    for (let i = 0; i < 12; i++) {
        id += chars[Math.floor(Math.random() * chars.length)];
    }
    return id;
}

function formatearArticulos(articulosStr) {
    if (!articulosStr) return 'Ninguno';
    
    const numeros = articulosStr.split(',').map(n => n.trim());
    const articulos = [];
    
    for (const num of numeros) {
        const articulo = ARTICULOS_MULTAS.find(a => a.numero === parseInt(num));
        if (articulo) {
            articulos.push(`Art.${articulo.numero} (${articulo.descripcion})`);
        } else {
            articulos.push(`Art.${num} (Desconocido)`);
        }
    }
    
    return articulos.join(', ');
}

// COMANDO: /multar
const multarCommand = {
    data: new SlashCommandBuilder()
        .setName('multar')
        .setDescription('Registra una multa a un usuario')
        .addUserOption(option =>
            option.setName('usuario')
                .setDescription('Usuario a multar')
                .setRequired(true))
        .addStringOption(option =>
            option.setName('articulos')
                .setDescription('Artículos de la multa (separados por coma) - Escribe para buscar')
                .setRequired(true)
                .setAutocomplete(true))
        .addIntegerOption(option =>
            option.setName('monto')
                .setDescription('Monto total de la multa (se calcula automáticamente)')
                .setRequired(false)),
    
    async autocomplete(interaction) {
        const focusedValue = interaction.options.getFocused();
        const opciones = [];
        
        for (const articulo of ARTICULOS_MULTAS) {
            const nombre = `Art.${articulo.numero} (${articulo.descripcion}) ($${articulo.monto})`;
            
            if (nombre.toLowerCase().includes(focusedValue.toLowerCase()) || 
                articulo.numero.toString().includes(focusedValue) ||
                articulo.descripcion.toLowerCase().includes(focusedValue.toLowerCase())) {
                
                opciones.push({
                    name: nombre,
                    value: articulo.numero.toString()
                });
                
                if (opciones.length >= 25) break; // Discord limita a 25 opciones
            }
        }
        
        await interaction.respond(opciones);
    },
    
    async execute(interaction) {
        await interaction.deferReply({ ephemeral: true });
        
        const usuario = interaction.options.getUser('usuario');
        const articulosInput = interaction.options.getString('articulos');
        const montoManual = interaction.options.getInteger('monto');
        
        // Validaciones
        if (usuario.bot) {
            return interaction.editReply({
                content: '❌ No puedes multar a un bot.',
                flags: MessageFlags.Ephemeral
            });
        }
        
        if (usuario.id === interaction.user.id) {
            return interaction.editReply({
                content: '❌ No puedes multarte a ti mismo.',
                flags: MessageFlags.Ephemeral
            });
        }
        
        // Procesar artículos
        const articulosNumeros = articulosInput.split(',').map(a => a.trim()).filter(a => a);
        
        if (articulosNumeros.length === 0) {
            return interaction.editReply({
                content: '❌ Debes especificar al menos un artículo válido.',
                flags: MessageFlags.Ephemeral
            });
        }
        
        // Calcular monto total
        let montoTotal = 0;
        const articulosValidos = [];
        
        for (const num of articulosNumeros) {
            const articuloNum = parseInt(num);
            const articulo = ARTICULOS_MULTAS.find(a => a.numero === articuloNum);
            
            if (articulo) {
                montoTotal += articulo.monto;
                articulosValidos.push(articulo.numero.toString());
            }
        }
        
        if (articulosValidos.length === 0) {
            return interaction.editReply({
                content: '❌ No se encontraron artículos válidos.',
                flags: MessageFlags.Ephemeral
            });
        }
        
        // Usar monto manual si se proporcionó
        if (montoManual !== null) {
            if (montoManual <= 0) {
                return interaction.editReply({
                    content: '❌ El monto debe ser mayor a 0.',
                    flags: MessageFlags.Ephemeral
                });
            }
            montoTotal = montoManual;
        }
        
        // Crear la multa
        const multa = {
            id: generarIdMulta(),
            usuarioId: usuario.id,
            oficialId: interaction.user.id,
            monto: montoTotal,
            articulos: articulosValidos.join(','),
            fechaCreacion: new Date().toISOString(),
            fechaLimite: new Date(Date.now() + 3 * 24 * 60 * 60 * 1000).toISOString(), // 3 días
            estado: 'PENDIENTE',
            cobradaAuto: false
        };
        
        // Guardar multa
        const multas = await leerMultas();
        multas.push(multa);
        await guardarMultas(multas);
        
        // Crear Container con la información de la multa
        const fechaTimestamp = Math.floor(new Date(multa.fechaLimite).getTime() / 1000);
        
        const container = new ContainerBuilder()
            .setAccentColor(0xFF0000) // Rojo para multa
            .addTextDisplayComponents(
                new TextDisplayBuilder()
                    .setContent('# 🚨 **MULTA REGISTRADA**')
            )
            .addSeparatorComponents(
                new SeparatorBuilder()
                    .setDivider(true)
                    .setSpacing(SeparatorSpacingSize.Small)
            )
            .addSectionComponents(
                new SectionBuilder()
                    .addTextDisplayComponents(
                        new TextDisplayBuilder()
                            .setContent(`**📋 ID de Multa:** \`${multa.id}\``),
                        new TextDisplayBuilder()
                            .setContent(`**👤 Usuario Multado:** ${usuario.username}`),
                        new TextDisplayBuilder()
                            .setContent(`**👮 Oficial:** ${interaction.user.username}`)
                    )
            )
            .addSectionComponents(
                new SectionBuilder()
                    .addTextDisplayComponents(
                        new TextDisplayBuilder()
                            .setContent(`**💰 Monto:** $${multa.monto}`),
                        new TextDisplayBuilder()
                            .setContent(`**⏰ Fecha Límite:** <t:${fechaTimestamp}:R>`),
                        new TextDisplayBuilder()
                            .setContent(`**📊 Estado:** ${multa.estado}`)
                    )
            )
            .addSeparatorComponents(
                new SeparatorBuilder()
                    .setDivider(true)
                    .setSpacing(SeparatorSpacingSize.Small)
            )
            .addTextDisplayComponents(
                new TextDisplayBuilder()
                    .setContent(`**📜 Artículos:** ${formatearArticulos(multa.articulos)}`)

                new TextDisplayBuilder()

                if (nuevaPagina.botones) {
                    updateOptions.components.push(nuevaPagina.botones);
                }
                
                await buttonInteraction.update(updateOptions);
                collector.resetTimer();
            });
            
            collector.on('end', () => {
                // Deshabilitar botones cuando expire
                if (botones) {
                    const botonesDeshabilitados = new ActionRowBuilder()
                        .addComponents(
                            new ButtonBuilder()
                                .setCustomId('multas_prev')
                                .setLabel('⬅️ Anterior')
                                .setStyle(ButtonStyle.Secondary)
                                .setDisabled(true),
                            new ButtonBuilder()
                                .setCustomId('multas_next')
                                .setLabel('Siguiente ➡️')
                                .setStyle(ButtonStyle.Secondary)
                                .setDisabled(true)
                        );
                    
                    mensaje.edit({
                        components: [container, botonesDeshabilitados],
                        flags: MessageFlags.IsComponentsV2
                    }).catch(() => {});
                }
            });
        }
    }
};

// COMANDO: /pagar-multa
const pagarMultaCommand = {
    data: new SlashCommandBuilder()
        .setName('pagar-multa')
        .setDescription('Paga una multa pendiente'),
    
    async execute(interaction) {
        await interaction.deferReply({ ephemeral: true });
        
        const multas = await leerMultas();
        const multasUsuario = multas.filter(m => 
            m.usuarioId === interaction.user.id && 
            m.estado === 'PENDIENTE'
        );
        
        if (multasUsuario.length === 0) {
            return interaction.editReply({
                content: 'No tienes multas pendientes a tu nombre.',
                flags: MessageFlags.Ephemeral
            });
        }
        
        // Calcular total a pagar
        const total = multasUsuario.reduce((sum, multa) => sum + multa.monto, 0);
        
        // Actualizar multas
        const ahora = new Date();
        for (const multa of multasUsuario) {
            multa.estado = 'PAGADA';
            multa.fechaPago = ahora.toISOString();
        }
        
        await guardarMultas(multas);
        
        // Crear Container de confirmación
        const container = new ContainerBuilder()
            .setAccentColor(0x00FF00) // Verde para pago exitoso
            .addTextDisplayComponents(
                new TextDisplayBuilder()
                    .setContent('# ✅ **PAGO REALIZADO**')
            )
            .addSeparatorComponents(
                new SeparatorBuilder()
                    .setDivider(true)
                    .setSpacing(SeparatorSpacingSize.Small)
            )
            .addSectionComponents(
                new SectionBuilder()
                    .addTextDisplayComponents(
                        new TextDisplayBuilder()
                            .setContent(`**👤 Usuario:** ${interaction.user.username}`),
                        new TextDisplayBuilder()
                            .setContent(`**📊 Multas Pagadas:** ${multasUsuario.length}`),
                        new TextDisplayBuilder()
                            .setContent(`**💰 Total Pagado:** $${total}`)
                    )
            )
            .addSeparatorComponents(
                new SeparatorBuilder()
                    .setDivider(true)
                    .setSpacing(SeparatorSpacingSize.Small)
            )
            .addTextDisplayComponents(
                new TextDisplayBuilder()
                    .setContent('_© Todos los derechos reservados, Chicago Real Life Roleplay | ER:LC | Spanish_')
            );
        
        await interaction.editReply({
            components: [container],
            flags: MessageFlags.IsComponentsV2 | MessageFlags.Ephemeral
        });
        
        console.log(`✅ ${interaction.user.username} pagó ${multasUsuario.length} multas por un total de $${total}`);
    }
};
